CREATE OR ALTER VIEW VSPED(
    COD_ESTAB,
    COD_DOC,
    COD_DOCIT,
    COD_PART,
    COD_ESTQ,
    COD_ITEM,
    DESCR_COMPL,
    CANC_DOC,
    CFOP_DOCIT,
    INTCFOP_DOCIT,
    COD_MOD,
    ESP_DOC,
    COD_SIT,
    DT_DOC,
    DT_E_S,
    IND_EMIT,
    IND_OPER,
    MOV_DOC,
    NUM_DOC,
    RAIZ_CNPJ,
    SER,
    CODCTD_PESS,
    UF_ESTB,
    COD_CONS,
    MOVESTQ_DOCIT,
    COD_NAT,
    COD_AIDF,
    ATIV2_ESTB,
    TP_ESC,
    ALIQPIS_ESTB,
    ALIQCOFINS_ESTB,
    CSNAEFD_ESTB,
    RECBRUTACS_ESTB,
    UN,
    CODBAR3_PROD,
    EX_IPI,
    COD_NCM,
    IND_MOV,
    FIN_CFOP,
    ESTAT_CFOP,
    NATBC_CS,
    TIPO_TRIB,
    DESC_CRED,
    COD_CTA,
    NAT_BC_CRED,
    SPED_PIS_COFINS,
    NAT_REC,
    TPEMI_DOC,
    FINNFE_DOC,
    STATUS_DOC,
    UF,
    ESTAB_COD_MUN,
    CNPJCPF_ESTB,
    PRINT_DOC)
AS
SELECT 
MV_SPED.COD_ESTAB, MV_SPED.COD_DOC, MV_SPED.COD_DOCIT, MV_SPED.COD_PART, MV_SPED.COD_ESTQ, MV_SPED.COD_ITEM, PROD.NOM_PROD DESCR_COMPL, MV_SPED.CANC_DOC, MV_SPED.CFOP_DOCIT, MV_SPED.INTCFOP_DOCIT, MV_SPED.COD_MOD, MV_SPED.ESP_DOC, MV_SPED.COD_SIT, MV_SPED.DT_DOC,
MV_SPED.DT_E_S, MV_SPED.IND_EMIT, MV_SPED.IND_OPER, MV_SPED.MOV_DOC, MV_SPED.NUM_DOC, SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) RAIZ_CNPJ, MV_SPED.SER, ESTAB.CODCTD_PESS, ESTAB.UF_ESTB, IIF(ESTAB.ATIV2_ESTB = "0", "04",  "") COD_CONS,
MV_SPED.MOVESTQ_DOCIT, MV_SPED.COD_NAT, MV_SPED.COD_AIDF, ESTAB.ATIV2_ESTB,
ESTAB.TPESCCSLP_ESTB TP_ESC, ESTAB.ALIQPIS_ESTB, ESTAB.ALIQCOFINS_ESTB, COALESCE(ESTAB.CSNAEFD_ESTB,"N")CSNAEFD_ESTB, ESTAB.RECBRUTACS_ESTB,
PROD.UNSD_PROD UN, PROD.CODBAR3_PROD, PROD.EXTIPI_PROD EX_IPI, PROD.TIPI_PROD COD_NCM, 
IIF(CFOP.MVFIS_CFOP = "N", 1, IIF(MV_SPED.MOVESTQ_DOCIT = "S",0, 1)) IND_MOV, COALESCE(CFOP.FIN_CFOP, "N") FIN_CFOP,
CFOP.ESTAT_CFOP ESTAT_CFOP,  
GRPIT.NATBC_CS, GRPIT.TIPO_TRIB, 
IIF(GRPIT.NATBC_CS = "13", COALESCE(SUBSTRING(CFOP.NOM_CFOP FROM 1 FOR 60), "OUTROS CREDITOS"), NULL) DESC_CRED,
IIF(CHAR_LENGTH(CFOP.COD_CTA) > 0, CFOP.COD_CTA, GRPIT.CODCTB_CTA) COD_CTA, 
IIF(MV_SPED.MOV_DOC = "E",RIGHT("00" || GRPIT.NATBC_CS, 2), NULL) NAT_BC_CRED,
COALESCE(GRPIT.SPEDPISCOFINS_TRIB,"N") SPED_PIS_COFINS,
IIF(CHAR_LENGTH(PROD.NATRECCS_PROD) > 0, PROD.NATRECCS_PROD, GRPIT.NATREC_TRIB) NAT_REC, MV_SPED.TPEMI_DOC, MV_SPED.FINNFE_DOC, MV_SPED.STATUS_DOC, MV_SPED.UF_PESS UF,
ESTAB.COD_MUN ESTAB_COD_MUN, ESTAB.CNPJCPF_ESTB, MV_SPED.PRINT_DOC
FROM (
SELECT 
DOC.COD_ESTAB, 
DOC.COD_DOC, 
COALESCE(DOCIT.COD_DOCIT,0) COD_DOCIT, 
IIF((DOC.CANC_DOC = "S" OR DOC.COD_MOD = "65" OR DOC.STATUS_DOC = "I"), NULL, DOC.COD_PESS) COD_PART,
DOCIT.COD_ESTQ, DOCIT.CODBAR_PROD COD_ITEM,
DOC.CANC_DOC, 
DOCIT.CFOP_DOCIT,
DOCIT.INTCFOP_DOCIT,
RIGHT("00" || DOC.COD_MOD,2) COD_MOD, DOC.ESP_DOC,
CASE
WHEN ((DOC.TIPO_DOC = "E") OR (DOC.TIPO_DOC = "E" AND DOC.FINNFE_DOC = "2")) THEN "01"
WHEN ((DOC.CANC_DOC = "S") AND (DOC.STATUS_DOC = "C")) THEN "02"
WHEN ((DOC.TIPO_DOC = "E") AND (DOC.CANC_DOC = "S")) THEN "03"
WHEN (DOC.STATUS_DOC = "I") THEN "05" WHEN DOC.STATUS_DOC = "D" THEN "04"
WHEN ((DOC.FINNFE_DOC = "2") AND (DOC.PRINT_DOC = "S") AND (DOC.CANC_DOC = "N")) THEN "06"
ELSE "00"
END COD_SIT,
IIF((DOC.CANC_DOC = "S"  OR DOC.STATUS_DOC = "I"), NULL, CAST(CAST(DOC.EMISS_DOC AS VARCHAR(11)) AS DATE)) DT_DOC,
IIF((DOC.CANC_DOC = "S"  OR DOC.STATUS_DOC = "I"), NULL, CAST(CAST(DOC.ENTSAID_DOC AS VARCHAR(11)) AS DATE)) DT_E_S,
IIF(((DOC.COD_AIDF > 0) OR (DOC.MOV_DOC = "S")), 0, 1) IND_EMIT,
IIF(DOC.MOV_DOC = "E", 0, 1) IND_OPER,
DOC.MOV_DOC,
DOC.NUM_DOC,
COALESCE(SUBSTRING(CHAVENFE_DOC FROM 23 FOR 3), RIGHT("000" || DOC.SERIE_DOC,3)) SER,
DOCIT.MOVESTQ_DOCIT,
DOCIT.CFOP_DOCIT || "-" || DOCIT.INTCFOP_DOCIT COD_NAT,
DOC.COD_AIDF, DOC.TPEMI_DOC, DOC.FINNFE_DOC, DOC.STATUS_DOC, DOC.UF_PESS, DOC.PRINT_DOC
FROM DOCIT
INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
WHERE
COALESCE(DOCIT.COD_DOCIT,0) < 990 AND DOC.CANC_DOC <> "S" AND COALESCE(DOCIT.CANC_DOCIT,"N") <> "S" AND COALESCE(DOCIT.DEL_DOCIT,"N") <> "S" AND
((DOC.COD_AIDF = 0 AND COALESCE(DOC.CANC_DOC, "N") = "N") OR DOC.COD_AIDF > 0)
AND ((DOC.COD_MOD IN ("1","1B","2","04","4","6","8","21","22","29","55","65","ISS", "NFSE")) or (DOC.MOV_DOC = "E" AND DOC.COD_MOD IN ("57","67")))
UNION ALL
SELECT 
DOC.COD_ESTAB, 
DOC.COD_DOC, 
1 COD_DOCIT,
DOC.CODF_PESS COD_PART,
-1 COD_ESTQ, -1 COD_ITEM,
DOC.CANC_DOC, 
DOC.CFOPS_DOC,
DOC.INTCFOP_DOC,
RIGHT("00" || DOC.COD_MOD,2) COD_MOD, DOC.ESP_DOC,
CASE
WHEN ((DOC.TIPO_DOC = "E") OR (DOC.TIPO_DOC = "E" AND DOC.FINNFE_DOC = "2")) THEN "01"
WHEN ((DOC.CANC_DOC = "S") AND (DOC.STATUS_DOC = "C")) THEN "02"
WHEN ((DOC.TIPO_DOC = "E") AND (DOC.CANC_DOC = "S")) THEN "03"
WHEN (DOC.STATUS_DOC = "I") THEN "05" WHEN DOC.STATUS_DOC = "D" THEN "04"
WHEN ((DOC.FINNFE_DOC = "2") AND (DOC.PRINT_DOC = "S") AND (DOC.CANC_DOC = "N")) THEN "06"
ELSE "00"
END COD_SIT,
IIF((DOC.CANC_DOC = "S"  OR DOC.STATUS_DOC = "I"), NULL, CAST(CAST(DOC.EMISS_DOC AS VARCHAR(11)) AS DATE)) DT_DOC,
IIF((DOC.CANC_DOC = "S"  OR DOC.STATUS_DOC = "I"), NULL, CAST(CAST(DOC.ENTSAID_DOC AS VARCHAR(11)) AS DATE)) DT_E_S,
0 IND_EMIT,
1 IND_OPER,
DOC.MOV_DOC,
DOC.NUM_DOC,
COALESCE(SUBSTRING(CHAVENFE_DOC FROM 23 FOR 3), RIGHT("000" || DOC.SERIE_DOC,3)) SER,
"N" MOVESTQ_DOCIT,
DOC.CFOPS_DOC || "-" || DOC.INTCFOP_DOC COD_NAT,
DOC.COD_AIDF, DOC.TPEMI_DOC, DOC.FINNFE_DOC, DOC.STATUS_DOC, DOC.UF_PESS, DOC.PRINT_DOC
FROM DOC
WHERE DOC.COD_AIDF > 0  AND DOC.MOV_DOC = "S" AND DOC.CANC_DOC <> "S" AND DOC.COD_MOD IN ("57","67")
) MV_SPED
INNER JOIN ESTAB ON ESTAB.COD_ESTAB = MV_SPED.COD_ESTAB
LEFT JOIN PESSOAS CTD ON CTD.COD_PESS = ESTAB.CODCTD_PESS
LEFT JOIN PROD ON (PROD.COD_ESTQ = MV_SPED.COD_ESTQ)
LEFT JOIN CFOP ON (CFOP.COD_CFOP = MV_SPED.CFOP_DOCIT AND CFOP.INT_CFOP = MV_SPED.INTCFOP_DOCIT)
LEFT JOIN GRP ON (GRP.COD_GRP = PROD.COD_GRP)
LEFT JOIN GRPIT ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP, 999) = GRPIT.COD_GRP AND MV_SPED.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
;
