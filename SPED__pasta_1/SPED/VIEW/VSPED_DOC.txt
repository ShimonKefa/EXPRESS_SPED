CREATE OR ALTER VIEW VSPED_DOC(
    REG,
    IND_OPER,
    IND_EMIT,
    COD_PART,
    COD_MOD,
    COD_SIT,
    SER,
    NUM_DOC,
    CHV_NFE,
    DT_DOC,
    DT_E_S,
    VL_DOC,
    IND_PGTO,
    VL_DESC,
    VL_ABAT_NT,
    VL_MERC,
    IND_FRT,
    VL_FRT,
    VL_SEG,
    VL_OUT_DA,
    VL_BC_ICMS,
    VL_ICMS,
    VL_BC_ICMS_ST,
    VL_ICMS_ST,
    VL_BC_IPI,
    VL_IPI,
    VL_BC_PIS,
    VL_PIS,
    VL_BC_COFINS,
    VL_COFINS,
    TP_CT_E,
    VL_NT,
    VL_ICMS_UF_REM,
    VL_ICMS_UF_DEST,
    VL_ISS,
    DIFAL,
    VL_FCP_UF_DEST,
    COD_MUN_ORIG,
    COD_MUN_DEST,
    COD_CTA,
    UF_ESTB,
    UF_PESS,
    CONTBICMS_PESS,
    CLTRIB3_DOC,
    CANC_DOC,
    CFOP,
    INTCFOP,
    CNPJ_CPF_PART,
    MOV_DOC,
    RAIZ_CNPJ,
    COD_CONS,
    COD_DOC,
    COD_ESTAB,
    ICMSA_DOC,
	PRINT_DOC,
	SPED_PIS_COFINS,
	ALIQ_ICMS,
	CST_ICMS)
AS
SELECT DISTINCT "DOCS" REG,

VSPED.IND_OPER,
VSPED.IND_EMIT,
VSPED.COD_PART,
VSPED.COD_MOD,
VSPED.COD_SIT,
VSPED.SER, 
DOC.NUM_DOC, 
IIF(VSPED.STATUS_DOC = "I", NULL, DOC.CHAVENFE_DOC) CHV_NFE, 
VSPED.DT_DOC, VSPED.DT_E_S,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, DOC.TOTALG_DOC) VL_DOC,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, CASE WHEN DOC.TOTFIN_DOC = 0 THEN "2" WHEN DOC.CPAGTO_DOC = "AV" THEN "0" WHEN DOC.CPAGTO_DOC = "AP" THEN "1" ELSE  "2" END)IND_PGTO,
DOC.VRDESC_DOC VL_DESC, 0.0 VL_ABAT_NT,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL , COALESCE(DOC.TOTPRODBR_DOC,0) + COALESCE(DOC.SERVISS_DOC,0) + COALESCE(DOC.ICMSR_DOC,0) + COALESCE(DOC.IPI_DOC,0) - IIF(COALESCE(VSPED.ATIV2_ESTB,"0") <> "0", 0.0, COALESCE(DOC.IPIE_DOC,0))) VL_MERC,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"),
NULL,
IIF(DOC.COD_MOD = "57",
IIF(COALESCE(DOC.MODFRETE_DOC,"9") IN ("3", "4"), "9", COALESCE(DOC.MODFRETE_DOC,"9")),
COALESCE(DOC.MODFRETE_DOC,"9")
)
)IND_FRT,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, COALESCE(DOC.FRETE_DOC,0))VL_FRT,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, COALESCE(DOC.SEGURO_DOC,0)) VL_SEG,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, COALESCE(DOC.OUTDESP_DOC,0)) VL_OUT_DA,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, COALESCE(DOC.BCICMSE_DOC,0)) VL_BC_ICMS,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, COALESCE(DOC.ICMSE_DOC,0)) VL_ICMS,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.BCICMSRE_DOC, 0)) VL_BC_ICMS_ST,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.ICMSRE_DOC,0)) VL_ICMS_ST,
IIF(((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65") OR COALESCE(VSPED.ATIV2_ESTB,"0") <> "0"), NULL , COALESCE(DOC.BCIPIE_DOC,0)) VL_BC_IPI,
IIF(((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65") OR COALESCE(VSPED.ATIV2_ESTB,"0") <> "0"), NULL , COALESCE(DOC.IPIE_DOC,0)) VL_IPI,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.BCPISE_DOC,0)) VL_BC_PIS, 
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.PISE_DOC,0)) VL_PIS,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.BCCOFINSE_DOC,0)) VL_BC_COFINS, 
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I" OR DOC.COD_MOD = "65"), NULL, COALESCE(DOC.COFINSE_DOC,0)) VL_COFINS,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC = "I"), NULL, DOC.FINNFE_DOC) TP_CT_E,
IIF((DOC.CANC_DOC = "S" OR DOC.STATUS_DOC  = "I"),NULL, (COALESCE(DOC.OUTROS_DOC,0) + COALESCE(DOC.ISENTO_DOC,0))) VL_NT,
IIF((DOC.UF_PESS = VSPED.UF_ESTB AND DOC.MOV_DOC = "E"), DOC.DALIQICMS_DOC, DOC.DALIQICMS_DOC - DOC.DALIQICMSD_DOC) VL_ICMS_UF_REM,
IIF((DOC.UF_PESS = VSPED.UF_ESTB AND DOC.MOV_DOC = "E"), 0.0, DOC.DALIQICMSD_DOC) VL_ICMS_UF_DEST, DOC.ISS_DOC VL_ISS,
DOC.DALIQICMS_DOC DIFAL, DOC.OUTRF_DOC VL_FCP_UF_DEST,
COALESCE(DOC.CODI_MUN, MUN_I.CODIBGE_MUN) COD_MUN_ORIG,
COALESCE(DOC.CODF_MUN, MUN_F.CODIBGE_MUN) COD_MUN_DEST,
NULL COD_CTA, VSPED.UF_ESTB, DOC.UF_PESS,DOC.CONTBICMS_PESS, DOC.CLTRIB3_DOC, DOC.CANC_DOC, DOC.CFOPS_DOC CFOP, DOC.INTCFOP_DOC INTCFOP, DOC.CNPJCPF_PESS CNPJ_CPF_PART, DOC.MOV_DOC,
VSPED.RAIZ_CNPJ, VSPED.COD_CONS, DOC.COD_DOC, DOC.COD_ESTAB,
DOC.ICMSA_DOC, VSPED.PRINT_DOC, VSPED.SPED_PIS_COFINS, DOC.ALIQAUTO_DOC ALIQ_ICMS, RIGHT('000' || COALESCE(DOC.CSTICMS_DOC,999),3) CST_ICMS
FROM DOC
INNER JOIN VSPED ON VSPED.COD_DOC = DOC.COD_DOC
LEFT JOIN MUN MUN_I ON (MUN_I.COD_MUN = IIF(DOC.MOV_DOC = "E" AND DOC.COD_MOD = "57", DOC.COD_MUN, VSPED.ESTAB_COD_MUN))
LEFT JOIN MUN MUN_F ON (MUN_F.COD_MUN = IIF(DOC.MOV_DOC = "E" AND DOC.COD_MOD = "57", VSPED.ESTAB_COD_MUN, DOC.COD_MUN))
