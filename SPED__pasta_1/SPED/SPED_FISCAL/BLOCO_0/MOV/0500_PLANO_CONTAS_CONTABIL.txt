1=SELECT DISTINCT '0500' REG,
2=CAST(:Data_Inicial AS DATE) DT_ALT, 
3=CTB_CONTA.GRUPO_CTA COD_NAT_CC, CTB_CONTA.TIPO_CTA IND_CTA, CTB_CONTA.NIVEL_CTA NIVEL, CTB_CONTA.CODCTB_CTA COD_CTA,
4=CTB_CONTA.NOM_CTA NOME_CTA FROM CTB_CONTA
5=INNER JOIN
6=(
7=SELECT DISTINCT CTB_CONTA.CODCTB_CTA FROM CIAPI
8=INNER JOIN CIAP ON CIAP.COD_CIAP = CIAPI.COD_CIAP
9=INNER JOIN PROD ON (PROD.COD_ESTQ = CIAPI.COD_ESTQ)
10=INNER JOIN CTB_CONTA ON (CTB_CONTA.COD_CTA = PROD.COD_CTA)
11=WHERE CIAPI.COD_ESTAB = :COD_ESTAB AND CIAP.DATAI_CIAP >= :Data_Inicial || ' 00:00:00' AND CIAP.DATAF_CIAP <= :Data_Final || ' 23:59:59'
12=UNION ALL
13=SELECT DISTINCT CTB_CONTA.CODCTB_CTA FROM DOCIT
14=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
15=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
16=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
17=INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
18=INNER JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
19=INNER JOIN CTB_CONTA ON (CTB_CONTA.CODCTB_CTA = COALESCE(GRPIT.CODCTB_CTA, CFOP.CODCTB_CTA))
20=WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S'  AND DOCIT.COD_DOCIT < 990
21=AND DOC.COD_ESTAB = :COD_ESTAB AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
22=AND DOC.PRINT_DOC = 'S' AND (DOC.COD_MOD IN ('1', '1B', '04', '4', '55') AND DOC.COD_AIDF =0)
23=UNION ALL
24=SELECT DISTINCT CFOP.CODCTB_CTA FROM DOC
25=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
26=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOC.CFOPS_DOC AND CFOP.INT_CFOP = DOC.INTCFOP_DOC)
27=WHERE DOC.COD_ESTAB = :COD_ESTAB: AND
28=((DOC.EMISS_DOC >= :DataHora_Inicial  AND DOC.EMISS_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'S') OR  (DOC.ENTSAID_DOC >= :DataHora_Inicial AND DOC.ENTSAID_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'E'))
29=AND DOC.COD_AIDF >= 0 AND DOC.COD_MOD IN ('21','22')
30=) MV ON (CTB_CONTA.CODCTB_CTA = MV.CODCTB_CTA)
31=WHERE CTB_CONTA.GRUPO_CTA = '1'
