[SQL]
1=SELECT DISTINCT '0300' REG, PROD.COD_ESTQ, PROD.CODBAR_PROD COD_IND_ITEM, CASE WHEN PROD.TIPO_PROD = 'B' THEN 1 ELSE '2' END IDENT_MERC, PROD.NOM_PROD DESCR_ITEM,
2=BEMPAI.CODBAR_PROD COD_PRNC, CTB_CONTA.CODCTB_CTA COD_CTA, PROD.ORIGEM_PROD NR_PARC
3=FROM PROD LEFT JOIN CTB_CONTA ON (CTB_CONTA.COD_CTA = COALESCE(PROD.COD_CTA,1))
4=INNER JOIN
5=(
6=SELECT DISTINCT PROD.COD_ESTQ FROM CIAPI
7=INNER JOIN PROD ON (PROD.COD_ESTQ = CIAPI.COD_ESTQ)
8=LEFT JOIN CIAP ON (CIAP.COD_CIAP = CIAPI.COD_CIAP)
9=WHERE CIAPI.COD_ESTAB = :COD_ESTAB
10=AND CIAPI.DATA_CIAPI >= :Data_Inicial AND CIAPI.DATA_CIAPI <= :Data_Final || ' 23:59:59' AND COALESCE(CIAPI.CANC_CIAPI,'N') = 'N'  AND COALESCE(CIAP.SIT_CIAP, "N") = "F"
11=) MV ON (PROD.COD_ESTQ = MV.COD_ESTQ)
12=LEFT JOIN PROD BEMPAI ON (BEMPAI.COD_ESTQ = PROD.CODMAST_ESTQ)
13=WHERE (PROD.COD_ESTAB = :COD_ESTAB)
14=EOF
