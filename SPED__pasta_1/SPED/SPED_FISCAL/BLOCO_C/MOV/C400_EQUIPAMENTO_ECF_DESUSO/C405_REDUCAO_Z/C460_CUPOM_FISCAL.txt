1=SELECT MAX(REG) REG, ':CRZ:' CRZ, MAX(COD_MOD) COD_MOD, MAX(DATA) DATA, MAX(DIA) DIA, MAX(ECF_FAB) ECF_FAB, MAX(COD_SIT) COD_SIT, MAX(NUM_DOC) NUM_DOC, MAX(DT_DOC) DT_DOC, MAX(VL_DOC) VL_DOC, SUM(VL_PIS) VL_PIS, SUM(VL_COFINS) VL_COFINS, MAX(CNPJCPF) CNPFCPF, MAX(NOM_ADQ) NOM_ADQ FROM
2=(
3=SELECT CUPOM.COD_CF, 'C460' REG, '2D' COD_MOD, CUPOM.DATA DATA, EXTRACT(DAY FROM CUPOM.DATA) DIA, CUPOM.SERIE ECF_FAB, CASE WHEN CUPOM.CANCELADO = 'S' THEN '02' ELSE '00' END COD_SIT,
4=CUPOM.NUMERO NUM_DOC, CASE WHEN CUPOM.CANCELADO = 'S' THEN NULL ELSE CUPOM.DATA END DT_DOC,
5=CASE WHEN CUPOM.CANCELADO = 'S' THEN NULL ELSE CASE WHEN COALESCE(CUPOM.DA,0) > 0 THEN (CUPOM.TOTAL + COALESCE(CUPOM.DA,0)) ELSE CUPOM.TOTAL END END VL_DOC, NULL VL_PIS,
6=NULL VL_COFINS, CASE WHEN CUPOM.CANCELADO = 'S' THEN NULL ELSE PESSOAS.CNPJCPF_PESS END CNPJCPF,
7=CASE WHEN CUPOM.CANCELADO = 'S' THEN NULL ELSE PESSOAS.NOM_PESS END NOM_ADQ FROM CUPOMIT
8=INNER JOIN CUPOM ON (CUPOMIT.COD_CF = CUPOM.COD_CF)
9=LEFT JOIN PESSOAS ON (PESSOAS.COD_PESS = CUPOM.COD_PESS)
10=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOMIT.COD_ESTAB)
11=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
12=INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
13=INNER JOIN ALIQECF ON (ALIQECF.ALIQUOTA = CUPOMIT.ALIQUOTA)
--14=LEFT JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
15=WHERE EXTRACT(DAY FROM CUPOM.DATA) = :DIA: AND CUPOM.SERIE = ':ECF_FAB:' AND CUPOM.DATA >= :Data_Inicial AND CUPOM.DATA <= :Data_Final AND
16=CUPOM.TIPO = 'CF'
17=) GROUP BY COD_CF
