[SQL]
1  = SELECT DISTINCT 'C120' REG, DOC.COD_DOC,
2  = '0' COD_DOC_IMP, DOCIT2.NUMDOC_DOCIT2 NUM_DOC_IMP, 
3  = ROUND(SUM(COALESCE(DOCIT.PIS_DOCIT,0)),2) VR_PIS_IMP,ROUND(SUM(COALESCE(DOCIT.COFINS_DOCIT,0)),2) VR_COFINS_IMP,
4  = DOCIT3.NUM_DOCIT3 NUM_ACDRAW
5  =  FROM DOCIT2 INNER JOIN DOC ON (DOC.COD_DOC = DOCIT2.COD_DOC) INNER JOIN DOCIT3 ON DOCIT2.COD_DOC = DOCIT3.COD_DOC AND DOCIT3.COD_DOCIT2 = DOCIT2.COD_DOCIT
6  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
7  =  INNER JOIN DOCIT ON (DOCIT.COD_DOC = DOCIT2.COD_DOC AND DOCIT.COD_DOCIT = DOCIT2.COD_DOCIT)
8  = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
9  = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
10 = LEFT JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
11 = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC: AND DOC.MOV_DOC = 'E' AND DOC.UF_PESS = 'EX' 
12  = AND DOC.COD_AIDF >= 0 
13  = AND DOC.COD_ESTAB = :COD_ESTAB AND ((DOC.EMISS_DOC >= :Data_Inicial AND DOC.EMISS_DOC <= :Data_Final AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial AND DOC.ENTSAID_DOC <= :Data_Final AND DOC.MOV_DOC = 'E'))
14 = GROUP BY REG, DOC.COD_DOC, COD_DOC_IMP, NUM_DOC_IMP, NUM_ACDRAW
