[SQL]
1=select 'C320' REG,
2=':DT_DOC:' DT_DOC,
3=CASE WHEN CHAR_LENGTH(COALESCE(DOCIT.CSTICMSE_DOCIT,0)) = 4 THEN SUBSTRING(COALESCE(DOCIT.CSTICMSE_DOCIT,0) FROM 2 FOR 4) ELSE COALESCE(DOCIT.CSTICMSE_DOCIT,0) END CST_ICMS,
4=DOCIT.CFOP_DOCIT cCFOP,
5=COALESCE(DOCIT.ALIQICMSE_DOCIT,0) ALIQ_ICMS,
6=SUM(DOCIT.TOTAL_DOCIT + COALESCE(DOCIT.IPI_DOCIT,0) + COALESCE(DOCIT.FRETE_DOCIT,0) + COALESCE(DOCIT.OUTDESP_DOCIT,0) + COALESCE(DOCIT.SEGURO_DOCIT,0) + COALESCE(DOCIT.ICMSR_DOCIT,0))  VL_OPR,
7=SUM(COALESCE(DOCIT.BCICMSE_DOCIT,0)) VL_BC_ICMS,
8=SUM(COALESCE(DOCIT.ICMSE_DOCIT,0)) VL_ICMS,
9=CASE WHEN SUM(DOCIT.REDBCICMS_DOCIT) > 0 THEN SUM(DOCIT.REDBCICMS_DOCIT) ELSE 0 END VL_RED_BC,
10='' COD_OBS
11=FROM DOCIT
12=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.cod_doc)
13=WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S'
14=AND DOCIT.COD_DOCIT < 990
15=AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.EMISS_DOC >= ':DT_DOC:' || ' 00:00:00' AND DOC.EMISS_DOC <= ':DT_DOC:' || ' 23:59:59' AND DOC.MOV_DOC = 'S'
16=AND DOC.COD_AIDF >=0 AND DOC.PRINT_DOC = 'S' AND DOC.COD_MOD IN ('2', '02') AND DOC.CANC_DOC <> 'S'
17=GROUP BY COALESCE(DOCIT.CSTICMSE_DOCIT,0), DOC.EMISS_DOC, DOCIT.CFOP_DOCIT, COALESCE(DOCIT.ALIQICMSE_DOCIT,0)
