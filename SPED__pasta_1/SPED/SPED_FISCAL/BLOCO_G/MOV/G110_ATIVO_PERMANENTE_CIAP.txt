[SQL]
1 = SELECT 'G110' REG, CIAP.COD_CIAP , CIAP.DATAI_CIAP DT_INI, CIAP.DATAF_CIAP DT_FIN, COALESCE(SALDO_CIAP.SALDO,0) SALDO_IN_ICMS, CIAP.VPARC_CIAP SOM_PARC, CIAP.TRBEXP_CIAP VL_TRIB_EXP,  
2 = CIAP.TSAIDA_CIAP VL_TOTAL, CIAP.INDICE_CIAP IND_PER_SAI, CIAP.CREDITO_CIAP ICMS_APROP, 0 SOM_ICMS_OC
3 = FROM CIAP
4 = LEFT JOIN (
5 = SELECT 
6 = CIAPI.COD_CIAP, SUM((COALESCE(PROD.VRATUAL_PROD, 0) + COALESCE(PROD.VRDN_PROD, 0) + COALESCE(PROD.VRDA_PROD, 0) + COALESCE(PROD.VRDF_PROD, 0))) SALDO
7 = FROM CIAPI
8 = INNER JOIN PROD ON CIAPI.COD_ESTQ = PROD.COD_ESTQ
9 = WHERE CIAPI.DATA_CIAPI BETWEEN :Data_Inicial AND :Data_Final  AND
10 = COALESCE(CIAPI.CANC_CIAPI, 'N') <> 'S' AND 
11 = CIAPI.COD_ESTAB = :COD_ESTAB AND CIAPI.NPARC_CIAPI <> 1
12 = GROUP BY COD_CIAP
13 = ) 
14 = SALDO_CIAP ON SALDO_CIAP.COD_CIAP = CIAP.COD_CIAP
15 = WHERE CIAP.DATAI_CIAP = :Data_Inicial AND CIAP.DATAF_CIAP = :Data_Final AND CIAP.COD_ESTAB = :COD_ESTAB AND CIAP.SIT_CIAP = 'F'
16 = EOF
