[SQL]
1=SELECT "K250" REG, DOC.EMISS_DOC DT_PROD,
2=PROD.CODBAR_PROD  COD_ITEM, SUM(DOCIT.QTDEESTQ_DOCIT) QTD
3=FROM DOCIT
4=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
5=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
6=WHERE CFOP.ESTAT_CFOP = "RET" AND DOC.EMISS_DOC >= CAST(:Data_Inicial AS DATE) AND DOC.EMISS_DOC <= CAST(:Data_Final AS DATE)
7=AND DOC.CANC_DOC = "N" AND DOC.PRINT_DOC = "S" AND DOC.MOV_DOC = 'E' AND ':IND_MOV:' = 0   AND CAST(:Data_Inicial AS DATE) >= '01/01/2017'
8=GROUP BY REG, DT_PROD, COD_ITEM, DOCIT.COD_ESTQ
