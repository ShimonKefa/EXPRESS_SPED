[SQL]
1 = SELECT 'K280' REG, INVIT.DTINVANT_PROD DT_EST, PROD.CODBAR_PROD COD_ITEM, 
2 = CASE WHEN INVIT.ESTQUNSD_PROD < 0 THEN ABS (INVIT.ESTQUNSD_PROD) END QTD_COR_NEG,
3 = CASE WHEN INVIT.ESTQUNSD_PROD >= 0 THEN INVIT.ESTQUNSD_PROD END QTD_COR_POS,
4 = CASE WHEN INVIT.CODPOSSE_PROD = 1 THEN 0 WHEN INVIT.CODPOSSE_PROD = 2 THEN 1 WHEN INVIT.CODPOSSE_PROD = 3 THEN 2 END IND_EST,
5 = IIF (INVIT.CODTERC_PESS = 0, NULL, INVIT.CODTERC_PESS) COD_PART
6 = FROM INVIT
7 = INNER JOIN PROD ON INVIT.COD_ESTQ = PROD.COD_ESTQ
8 = WHERE COALESCE(GRADEC_PROD,'') = 'AJ' AND INVIT.COD_ESTQ = :COD_ESTQ: AND CASE WHEN INVIT.CODPOSSE_PROD = 1 THEN 0 WHEN INVIT.CODPOSSE_PROD = 2 THEN 1 WHEN INVIT.CODPOSSE_PROD = 3 THEN 2 END = :IND_EST: AND :PARAM_7 = 'S' 
9 = AND DTINV_PROD = CAST(:PARAM_8 AS DATE) AND INVIT.CODTERC_PESS = :COD_PART2:
