1 = SELECT '1600' REG, COD_PART, SUM(TOT_CREDITO) TOT_CREDITO, SUM(TOT_DEBITO) TOT_DEBITO FROM (
2 = SELECT '1600' REG, COBRANCA.CODF_PESS COD_PART, CASE COBRANCA.FINNFCE_COBR WHEN 3 THEN DOCX.VALOR_DCX ELSE 0 END TOT_CREDITO , 
3 = CASE COBRANCA.FINNFCE_COBR WHEN 4 THEN DOCX.VALOR_DCX ELSE 0 END TOT_DEBITO FROM DOCX
4 = INNER JOIN DOC ON DOC.COD_DOC = DOCX.COD_DOC
5 = INNER JOIN COBRANCA ON COBRANCA.COD_COBR = DOCX.COD_COBR
6 = WHERE DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.PRINT_DOC = 'S' 
7 = AND DOC.CANC_DOC = 'N' AND DOC.COD_AIDF > 0 AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.MOV_DOC = 'S' AND COBRANCA.FINNFCE_COBR IN (3, 4)
8 = UNION ALL
9 = SELECT '1600' REG, COBRANCA.CODF_PESS COD_PART, CASE COBRANCA.FINNFCE_COBR WHEN 3 THEN FATURAS.SALDO_FAT ELSE 0 END TOT_CREDITO, 
10 = CASE COBRANCA.FINNFCE_COBR WHEN 4 THEN FATURAS.SALDO_FAT ELSE 0 END TOT_DEBITO FROM FATURAS
11 = INNER JOIN DOC ON DOC.COD_FAT = FATURAS.COD_FAT
12 = INNER JOIN COBRANCA ON COBRANCA.COD_COBR = FATURAS.COD_COBR
13 = LEFT JOIN DOCX ON DOC.COD_DOC = DOCX.COD_DOC
14 = WHERE DOCX.COD_DOC IS NULL AND DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.PRINT_DOC = 'S' 
15 = AND DOC.CANC_DOC = 'N' AND DOC.COD_AIDF > 0 AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.MOV_DOC = 'S' AND COBRANCA.FINNFCE_COBR IN (3, 4)
16 = UNION ALL
17 = SELECT '1600' REG, COBRANCA.CODF_PESS COD_PART,  CASE COBRANCA.FINNFCE_COBR WHEN 3 THEN FATURBX.SALDO_FAT ELSE 0 END TOT_CREDITO, 
18 = CASE COBRANCA.FINNFCE_COBR WHEN 4 THEN FATURBX.SALDO_FAT ELSE 0 END TOT_DEBITO FROM FATURBX
19 = INNER JOIN DOC ON DOC.COD_FAT = FATURBX.COD_FAT
20 = INNER JOIN COBRANCA ON COBRANCA.COD_COBR = FATURBX.COD_COBR
21 = LEFT JOIN DOCX ON DOC.COD_DOC = DOCX.COD_DOC
22 = WHERE DOCX.COD_DOC IS NULL AND DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.PRINT_DOC = 'S' 
23 = AND DOC.CANC_DOC = 'N' AND DOC.COD_AIDF > 0 AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.MOV_DOC = 'S' AND COBRANCA.FINNFCE_COBR IN (3, 4)
24 = ) WHERE CAST(:Data_Final AS DATE) <= '12/31/2021' AND REG = '1600'
25 = GROUP BY REG, COD_PART
