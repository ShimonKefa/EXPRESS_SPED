[SQL]
1=SELECT DISTINCT '1400' REG, 
2=CASE 
3=WHEN FLAG = 'FLAG1' AND COALESCE(ESTAB.VERSAOSIS_ESTB, 'INTERNO') = 'TRANSP' THEN 'CTE' 
4=WHEN FLAG = 'FLAG2' THEN 'NFE' 
5=WHEN FLAG = 'FLAG3' THEN 'BAS21'
6=END COD_ITEM, 
7=CASE WHEN (MUN_ESTB.SIGLA_UF IN ('ES', 'MG', 'BA') AND DOC.UF_PESS = ESTAB.UF_ESTB) OR (DOC.COD_MOD = '55') THEN MUN.CODIBGE_MUN ELSE MUN_ESTB.CODIBGE_MUN END COD_MUN, 
8=SUM(TOTALG_DOC) VR_MENSAL
9=FROM (
10=SELECT 'FLAG1' FLAG, DOC.COD_DOC, DOC.COD_AIDF, DOC.PRINT_DOC, DOC.CANC_DOC, DOC.EMISS_DOC, DOC.COD_ESTAB, DOC.COD_MOD, DOC.CODI_MUN, DOC.UFI_MUN UF_PESS, DOC.TOTALG_DOC FROM DOC
11=INNER JOIN ESTAB ON ESTAB.COD_ESTAB = DOC.COD_ESTAB 
12=WHERE DOC.COD_MOD = '57' AND DOC.MOV_DOC = 'S' AND ESTAB.UF_ESTB <> 'BA' 
13=UNION ALL
14=SELECT 'FLAG2' FLAG, DOC.COD_DOC, MAX(DOC.COD_AIDF) COD_AIDF, MAX(DOC.PRINT_DOC) PRINT_DOC, MAX(DOC.CANC_DOC) CANC_DOC, MAX(DOC.EMISS_DOC) EMISS_DOC, MAX(DOC.COD_ESTAB) COD_ESTAB, MAX(DOC.COD_MOD) COD_MOD, MAX(DOC.COD_MUN) CODI_MUN, MAX(DOC.UF_PESS) UF_PESS, MAX(DOC.TOTALG_DOC) FROM DOCIT 
15=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) 
16=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
17=WHERE DOC.MOV_DOC = 'E' AND DOC.COD_AIDF > 0 AND COALESCE(CFOP.ESTAT_CFOP,"") = 'COMPRAS' AND DOC.FINNFE_DOC IN ('1','2')
18=GROUP BY DOC.COD_DOC
19=UNION ALL
20=SELECT 'FLAG3' FLAG, DOC.COD_DOC, DOC.COD_AIDF, DOC.PRINT_DOC, DOC.CANC_DOC, DOC.EMISS_DOC, DOC.COD_ESTAB, DOC.COD_MOD, DOC.CODI_MUN, DOC.UFI_MUN UF_PESS, DOC.TOTALG_DOC FROM DOC
21=INNER JOIN ESTAB ON ESTAB.COD_ESTAB = DOC.COD_ESTAB 
22=WHERE DOC.COD_MOD = '57' AND DOC.MOV_DOC = 'S' AND ESTAB.UF_ESTB = 'BA'
23=) DOC
24=INNER JOIN MUN ON MUN.COD_MUN = DOC.CODI_MUN
25=INNER JOIN ESTAB ON ESTAB.COD_ESTAB = DOC.COD_ESTAB
26=INNER JOIN MUN MUN_ESTB ON MUN_ESTB.COD_MUN = ESTAB.COD_MUN
27=WHERE :CONT: > 0 AND DOC.EMISS_DOC BETWEEN :Data_Inicial || ' 00:00:00' AND :Data_Final || ' 23:59:59' AND DOC.COD_ESTAB = :COD_ESTAB AND COALESCE(ESTAB.VERSAOSIS_ESTB, 'INTERNO') = 'TRANSP'
28=AND DOC.COD_AIDF > 0 AND DOC.PRINT_DOC = 'S' AND DOC.CANC_DOC = 'N' 
29=GROUP BY COD_MOD, COD_MUN, VERSAOSIS_ESTB, ESTAB.UF_ESTB, FLAG
