1=SELECT DISTINCT 'C113' REG, DOCREF.COD_DOC COD_DOC, CASE WHEN DOCREF2.MOV_DOC = 'E' AND DOCREF2.COD_AIDF = 0 THEN 0 ELSE 1 END IND_OPER, 
2=CASE WHEN DOCREF2.MOV_DOC = 'E' AND DOCREF2.COD_AIDF = 0 THEN 1 ELSE 0 END  IND_EMIT,
3=CASE WHEN ((DOC.CANC_DOC = 'S') OR (DOCREF.COD_MOD = '65')) THEN NULL ELSE DOCREF2.COD_PESS END COD_PART, CASE WHEN DOCREF2.COD_MOD IN ('1','1A') THEN '01' ELSE COALESCE(DOCREF2.COD_MOD,'55') END COD_MOD,
4=DOCREF2.SERIE_DOC SER, '' SUB, DOCREF2.NUM_DOC,
5=CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOCREF2.EMISS_DOC END DT_DOC,
6=DOCREF.CHAVENFE_DOC FROM DOCREF
7=INNER JOIN DOC ON (DOC.COD_DOC = DOCREF.COD_DOC)
8=INNER JOIN PESSOAS ON PESSOAS.CNPJCPF_PESS = SUBSTRING(DOCREF.CHAVENFE_DOC FROM 7 FOR 14)
9=INNER JOIN DOC DOCREF2 ON DOCREF2.CHAVENFE_DOC = DOCREF.CHAVENFE_DOC AND DOCREF2.COD_ESTAB = :COD_ESTAB
10=WHERE  TRIM(DOCREF.CHAVENFE_DOC) <> '' AND DOCREF.TIPO_DOC = 'E' AND
11=COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_ESTAB = :COD_ESTAB AND 
12=((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
13=AND DOC.PRINT_DOC = 'S' AND (DOC.COD_MOD IN ('1', '1B', '04', '4', '55') AND DOC.COD_AIDF >= 0) AND DOCREF2.COD_AIDF >= 0 AND DOC.STATUS_DOC NOT IN ('I', 'D') AND DOCREF2.COD_MOD IN ('1', '1B', '04', '4', '55')
14 = EOF
