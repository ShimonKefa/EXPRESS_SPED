[SQL]
1=SELECT 'C101' REG, DOC.COD_DOC, 0.0 VL_FCP_UF_DEST, 
2= 0.0 VL_ICMS_UF_DEST, 
3= CASE WHEN DOC.UF_PESS = ESTAB.UF_ESTB AND DOC.MOV_DOC = 'E' THEN DALIQICMS_DOC ELSE DALIQICMS_DOC - DALIQICMSD_DOC END VL_ICMS_UF_REM, CASE WHEN DOC.MOV_DOC = 'E' THEN 0 ELSE 1 END IND_OPER, ESTAB.UF_ESTB,
4=CASE
5=WHEN DOC.FINNFE_DOC = '2' THEN '06' -- DOC.FISCAL COMPLEMENTAR
6=WHEN DOC.TIPO_DOC = 'E' THEN '01'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
7=WHEN DOC.CANC_DOC = 'S' THEN '02' -- DOCUMENTO CANCELADO
8=WHEN DOC.TIPO_DOC = 'E' AND DOC.CANC_DOC = 'S' THEN '03'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
9=WHEN DOC.TIPO_DOC = 'E' AND DOC.FINNFE_DOC = '2' THEN '01'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
10=WHEN DOC.STATUS_DOC = 'I' THEN '05'
11=ELSE '00'  --DOCUMENTO REGULAR
12=END COD_SIT, ESTAB.UF_ESTB UF_ESTB
13=FROM DOC INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
14=WHERE (DALIQICMS_DOC > 0) AND COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC:
15=AND DOC.COD_AIDF > 0
16=AND DOC.COD_ESTAB = :COD_ESTAB AND (DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S')
17=AND DOC.COD_MOD = '57'
18=AND DOC.CONTBICMS_PESS = 'N' AND DOC.CLTRIB3_DOC = '1' AND DOC.UF_PESS <> 'EX'
