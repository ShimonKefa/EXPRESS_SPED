[SQL]
1=SELECT 'A001' REG, CASE WHEN MV.CONT > 0 THEN '0' ELSE '1' END IND_MOV, COD_ESTAB, SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) RAIZ_CNPJ,
2=ESTAB.TPESCCS_ESTB IND_ESCRI, ESTAB.APUCRED1_ESTB APU_CRED,
3=ESTAB.INCTRIB1_ESTB COD_INC_TRIB FROM ESTAB
4=INNER JOIN
5=(
6=SELECT MAX(COD_ESTAB) COD_ESTAB, SUM(CONT) CONT FROM
7=(
8=SELECT :COD_ESTAB: COD_ESTAB, COUNT(*) CONT FROM DOC
9=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
10=WHERE (
11=((COALESCE(DOC.MOV_DOC,'SM') IN ('S', 'SM')) AND (DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00') AND (DOC.EMISS_DOC <= :Data_Final || ' 23:59:59'))  OR
12=((DOC.MOV_DOC='E') AND (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00') AND (DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59'))
13=) AND
14=((DOC.ESP_DOC = 'ISS') OR DOC.COD_MOD IN ('NFSE', 'ISS')) AND (DOC.PRINT_DOC = 'S') AND (DOC.COD_AIDF >=0) AND
15=(COALESCE(DOC.TPAMB_DOC,'1') <> '2') AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ:' AND
16=(
17=':IND_ESCRI:' = '2' AND ((':COD_INC_TRIB:' <> '2' AND DOC.MOV_DOC = 'E') OR (DOC.MOV_DOC <> 'E'))
18=)
19=)
20=) MV ON (ESTAB.COD_ESTAB = MV.COD_ESTAB)
21=WHERE ESTAB.COD_ESTAB = :COD_ESTAB
