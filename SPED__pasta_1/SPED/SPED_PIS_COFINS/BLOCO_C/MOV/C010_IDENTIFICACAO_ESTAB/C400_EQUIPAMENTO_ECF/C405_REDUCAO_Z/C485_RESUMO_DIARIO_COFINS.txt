[SQL]
1=SELECT 'C485' REG, ECF_FAB,  DATA DT_DOC,  CST_COFINS, SUM( VL_ITEM) VL_ITEM, SUM( VL_BC_COFINS) VL_BC_COFINS,ALIQ_COFINS,MAX( QUANT_BC_COFINS) QUANT_BC_COFINS, MAX( ALIQ_COFINS_QUANT) ALIQ_COFINS_QUANT,
2=SUM(TRUNC(VL_COFINS,5)) VL_COFINS,  COD_ITEM, MAX( COD_CTA) COD_CTA, ':IND_ESCRI:' IND_ESCRI  FROM (
3=SELECT 'C485' REG, GRPIT.CST_COFINS, CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0) VL_ITEM,
4=CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
5=0
6=ELSE
7=CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
8=((CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0))) -
9=((CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0)) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)
10=ELSE
11=((CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0))) -
12=((CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0)) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)
13=END
14=END VL_BC_COFINS,
15=COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) ALIQ_COFINS,
16='' QUANT_BC_COFINS, '' ALIQ_COFINS_QUANT,
17=CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
18=0
19=ELSE
20=(CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0) - (CUPOMIT.TOTAL * COALESCE(GRPIT.RDBC_PIS,0) / 100))  * COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
21=END VL_COFINS, CUPOM.SERIE ECF_FAB,
22=PROD.CODBAR_PROD COD_ITEM, COALESCE(CFOP.CODCTB_CTA, GRPIT.CODCTB_CTA) COD_CTA, CUPOMIT.DATA FROM CUPOMIT
23=INNER JOIN CUPOM ON (CUPOMIT.COD_CF = CUPOM.COD_CF)
24=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOMIT.COD_ESTAB)
25=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
26=INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
27=INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
28=WHERE CUPOMIT.DATA = ":DT_DIA:" AND CUPOMIT.SERIE = ':ECF_FAB:' AND CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final
29=AND COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N') AND ESTAB.COD_ESTAB = :COD_ESTAB:
30=) GROUP BY CST_COFINS, ALIQ_COFINS, COD_ITEM,  DATA, ECF_FAB
