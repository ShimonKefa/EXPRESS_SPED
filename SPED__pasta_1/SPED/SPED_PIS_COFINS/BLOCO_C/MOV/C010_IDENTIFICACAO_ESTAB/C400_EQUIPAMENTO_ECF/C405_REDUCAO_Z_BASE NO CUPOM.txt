3  = (
4  = SELECT DISTINCT

14 = CASE WHEN CUPOMIT.ALIQUOTA="ACRE" THEN
15 =  0
16 = ELSE
17 = SUM(CUPOMIT.TOTAL + CUPOMIT.DACUPOM)
18 = END VL_BRT,
19 = MAX(COALESCE(CUPOMIT.GT,0)) GT
20 = FROM CUPOMIT INNER JOIN CUPOM ON (CUPOM.COD_CF = CUPOMIT.COD_CF)
21 = LEFT JOIN REDZECF ON (REDZECF.SERIE = CUPOMIT.SERIE AND REDZECF.MOVIMENTO = CUPOMIT.DATA)
22 = WHERE
23 = CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final AND CUPOMIT.SERIE = ':ECF_FAB:' AND CUPOMIT.COD_ESTAB = :COD_ESTAB: AND (CUPOM.CANCELADO = 'N' AND CUPOMIT.CANCELADO = 'N')

25 = ) MV
