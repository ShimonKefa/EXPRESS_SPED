[SQL]
1 = SELECT DISTINCT 'C110' REG, COD_DOC, COD_INF, MAX(TXT_COMPL) FROM (
2 = SELECT DISTINCT 'C110' REG, DOCIT.COD_DOC, DOCIT.PEDRESP_DOCIT COD_INF, 'NOTA DE DEVOLUCAO N ' ||  DOCDEV.NUM_DOC TXT_COMPL FROM DOCIT
3 = INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
4 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
5 = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
6 = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
7 = INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
8 = LEFT JOIN DOC DOCDEV ON DOCDEV.COD_DOC = DOCIT.CODREF_DOC
11 = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC: AND DOC.MOV_DOC = 'S' AND COALESCE(DOCIT.PEDRESP_DOCIT,'NAO') = 'DEV'
12 = AND DOC.COD_AIDF >= 0 
13 = AND DOC.COD_ESTAB = :COD_ESTAB: AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
14 = AND COALESCE(GRPIT.SPEDPISCOFINS_TRIB,'N') = 'S'  AND DOC.COD_MOD IN ('1', '1B', '04', '4', '55') AND :IND_ESCRI: = 2 
15 = UNION ALL
16 = SELECT DISTINCT 'C110' REG, DOC.COD_DOC, 'INFFIS' COD_INF, CASE WHEN SUBSTRING(DOC.OBS_DOC FROM 256 FOR 10) <> '' THEN SUBSTRING(DOC.OBS_DOC FROM 1 FOR 200)||'...' ELSE DOC.OBS_DOC END TXT_COMPL FROM DOC
17 = INNER JOIN DOCIT ON (DOC.COD_DOC = DOCIT.COD_DOC)
18 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
19 = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
20 = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
21 = INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
22 = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC: AND DOC.MOV_DOC = 'S' AND COALESCE(DOC.OBS_DOC,'NAO') <> 'NAO'
23 = AND DOC.COD_AIDF >= 0
24 = AND DOC.COD_ESTAB = :COD_ESTAB: AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
25 = AND COALESCE(GRPIT.SPEDPISCOFINS_TRIB,'N') = 'S'  AND DOC.COD_MOD IN ('1', '1B', '04', '4', '55') AND :IND_ESCRI: = 2 
26 = UNION ALL
27 = SELECT DISTINCT 'C110' REG, DOC.COD_DOC, 'INFCON' COD_INF, CASE WHEN SUBSTRING(DOC.OBS2_DOC FROM 256 FOR 10) <> '' THEN SUBSTRING(DOC.OBS2_DOC FROM 1 FOR 200)||'...' ELSE DOC.OBS2_DOC END TXT_COMPL FROM DOC
28 = INNER JOIN DOCIT ON (DOC.COD_DOC = DOCIT.COD_DOC)
29 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
30 = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
31 = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
32 = INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
33 = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC: AND DOC.MOV_DOC = 'S' AND COALESCE(DOC.OBS2_DOC,'NAO') <> 'NAO'
34 = AND DOC.COD_AIDF >= 0 
35 = AND DOC.COD_ESTAB = :COD_ESTAB: AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
36 = AND COALESCE(GRPIT.SPEDPISCOFINS_TRIB,'N') = 'S' AND DOC.COD_MOD IN ('1', '1B', '04', '4', '55') AND :IND_ESCRI: = 2 
37 = ) GROUP BY COD_DOC, COD_INF
