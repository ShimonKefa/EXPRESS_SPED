[SQL]
1=SELECT 'D001' REG, CASE WHEN MV.CONT > 0 THEN '0' ELSE '1' END IND_MOV, COD_ESTAB, SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) RAIZ_CNPJ,
2=ESTAB.TPESCCS_ESTB IND_ESCRI, ESTAB.INCTRIB1_ESTB COD_INC_TRIB FROM ESTAB
3=INNER JOIN
4=(
5=SELECT MAX(COD_ESTAB) COD_ESTAB, SUM(CONT) CONT FROM
6=(
7=SELECT :COD_ESTAB: COD_ESTAB, COUNT(*) CONT FROM DOC
8=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
9=WHERE (
10=((DOC.MOV_DOC='S') AND (DOC.EMISS_DOC >= :DataHora_Inicial) AND (DOC.EMISS_DOC <= :DataHora_Final))  OR
11=((DOC.MOV_DOC='E') AND (DOC.ENTSAID_DOC >= :DataHora_Inicial) AND (DOC.ENTSAID_DOC <= :DataHora_Final))
12=) AND
13=((DOC.ESP_DOC = 'CT') OR (DOC.COD_MOD IN ('21','22'))) AND (DOC.PRINT_DOC = 'S') AND (DOC.COD_AIDF >=0) AND
14=(COALESCE(DOC.TPAMB_DOC,'1') <> '2') AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ:'
15=)
16=) MV ON (ESTAB.COD_ESTAB = MV.COD_ESTAB)
17=WHERE ESTAB.COD_ESTAB = :COD_ESTAB
