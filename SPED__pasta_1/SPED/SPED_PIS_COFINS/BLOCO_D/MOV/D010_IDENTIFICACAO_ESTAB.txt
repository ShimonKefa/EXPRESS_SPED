[SQL]
1=SELECT DISTINCT 'D010' REG, ESTAB.CNPJCPF_ESTB CNPJ,
2=ESTAB.COD_ESTAB COD_ESTAB, ':IND_ESCRI:' IND_ESCRI, ':COD_INC_TRIB:'  COD_INC_TRIB FROM ESTAB INNER JOIN  DOC ON (DOC.COD_ESTAB = ESTAB.COD_ESTAB)
3=LEFT JOIN MUN ON (MUN.COD_MUN = ESTAB.COD_MUN)
4=WHERE NOT (COALESCE(ESTAB.IDPIS_ESTB,'') = '21' OR COALESCE(ESTAB.IDCOFINS_ESTB,'') = '21') AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ:' AND
5=((DOC.EMISS_DOC >= :DataHora_Inicial AND DOC.EMISS_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'S')
6=OR
7=(DOC.ENTSAID_DOC >= :DataHora_Inicial AND DOC.ENTSAID_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'E'))
8=AND (DOC.COD_MOD IN ('21','22','8','57','67')) AND ':IND_MOV:' <> '1'
