[SQL]
1 = SELECT DISTINCT 'I051' REG, LEFT(CTB_CONTA.CODCTB_CTA || '                    ', 20) ORDEM, --'2' COD_PLAN_REF, 
2 = '' COD_CCUS, CTB_CONTARF.CODCTB_CTARF COD_CTA_REF FROM CTB_CONTA INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CTB_CONTA.COD_ESTAB)
3 = INNER JOIN CTB_CONTARF ON (CTB_CONTARF.CODCTB_CTARF = CTB_CONTA.CODCTB2_CTA)
4 = INNER JOIN (SELECT D_CODCTB_CTA CODCTB_CTA, DATA_CTB FROM CTB_MOV UNION ALL SELECT C_CODCTB_CTA CODCTB_CTA, DATA_CTB FROM CTB_MOV) MV ON (MV.CODCTB_CTA = CTB_CONTA.CODCTB_CTA)
5 = WHERE ESTAB.CTRLCTB_ESTB  = :CTRLCTB_ESTB AND CTB_CONTA.SIT_CTA <> 'I' AND CTB_CONTA.TIPO_CTA = 'A' AND CTB_CONTA.COD_CTA = ":ORDEM:" AND MV.DATA_CTB >= :Data_Inicial AND MV.DATA_CTB <= :Data_Final
