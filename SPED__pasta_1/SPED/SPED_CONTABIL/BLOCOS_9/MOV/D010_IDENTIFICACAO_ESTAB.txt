[SQL]
1 = SELECT DISTINCT 'D010' REG, ESTAB.COD_ESTAB COD_ESTAB, 
2 = ESTAB.CNPJCPF_ESTB CNPJ FROM ESTAB INNER JOIN DOC ON (DOC.COD_ESTAB = ESTAB.COD_ESTAB)
3 = LEFT JOIN MUN ON (MUN.COD_MUN = ESTAB.COD_MUN) 
4 = WHERE SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ:' AND 
5 = ((DOC.EMISS_DOC >= :Data_Inicial AND DOC.EMISS_DOC <= :Data_Final AND DOC.MOV_DOC = 'S')
6 =  OR
7 =  (DOC.ENTSAID_DOC >= :Data_Inicial AND DOC.ENTSAID_DOC <= :Data_Final AND DOC.MOV_DOC = 'E'))
8 =   AND DOC.ESP_DOC IN ('CT')
