1=SELECT DISTINCT 'D500' REG, 0 IND_OPER, 1 IND_EMIT, DOC.COD_DOC,
2=DOC.COD_PESS COD_PART, DOC.COD_MOD, '00' COD_SIT,
3=DOC.SERIE_DOC SER, DOC.SUBSERIE_DOC SUB, DOC.NUM_DOC NUM_DOC,
4=DOC.EMISS_DOC DT_DOC, DOC.ENTSAID_DOC DT_A_P,
5=DOC.TOTALG_DOC VL_DOC, DOC.VRDESC_DOC VL_DESC, TOTPRODBR_DOC VL_SERV, TOTSERVBR_DOC VL_SERV_NT, 0.0 VL_TERC, DOC.OUTDESP_DOC VL_DA,
6=DOC.BCICMSE_DOC VL_BC_ICMS, DOC.ICMSE_DOC VL_ICMS, '' COD_INF, '' VL_PIS, '' VL_COFINS, COALESCE(CFOP.COD_CTA, DOCIT.CHAVE_DOCIT) COD_CTA, 1 TP_ASSINANTE
7=FROM DOC
8=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
9=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOC.CFOPS_DOC AND CFOP.INT_CFOP = DOC.INTCFOP_DOC)
10=INNER JOIN DOCIT ON (DOC.COD_DOC = DOCIT.COD_DOC)
11=WHERE DOC.COD_ESTAB = :COD_ESTAB AND
12=((DOC.EMISS_DOC >= :DataHora_Inicial  AND DOC.EMISS_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'S') OR  (DOC.ENTSAID_DOC >= :DataHora_Inicial AND DOC.ENTSAID_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'E'))
13=AND DOC.COD_AIDF >= 0 AND DOC.COD_MOD IN ('21','22')
