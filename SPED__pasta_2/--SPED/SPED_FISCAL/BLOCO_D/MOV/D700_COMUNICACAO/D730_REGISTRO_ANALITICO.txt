[SQL]
1=SELECT 'D730' REG, DOC.COD_DOC,
2=CASE WHEN CHAR_LENGTH(COALESCE(DOCIT.CSTICMSE_DOCIT,0)) = 4 THEN SUBSTRING(COALESCE(DOCIT.CSTICMSE_DOCIT,0) FROM 2 FOR 4) ELSE COALESCE(DOCIT.CSTICMSE_DOCIT,0) END CST_ICMS,
3=DOCIT.CFOP_DOCIT CFOP, DOCIT.ALIQICMS_DOCIT ALIQ_ICMS,
4=SUM(DOCIT.TOTAL_DOCIT + COALESCE(DOCIT.IPI_DOCIT,0) + COALESCE(DOCIT.FRETE_DOCIT,0) + COALESCE(DOCIT.OUTDESP_DOCIT,0) + COALESCE(DOCIT.SEGURO_DOCIT,0) + COALESCE(DOCIT.ICMSRE_DOCIT,0))  VL_OPR,
5=SUM(COALESCE(DOCIT.BCICMSE_DOCIT,0)) VL_BC_ICMS, SUM(COALESCE(DOCIT.ICMSE_DOCIT,0)) VL_ICMS,
6=CASE WHEN SUM(COALESCE(DOCIT.REDBCICMS_DOCIT,0)) > 0 THEN SUM(COALESCE(DOCIT.REDBCICMS_DOCIT,0)) ELSE 0 END VL_RED_BC, '' COD_OBS
7=FROM DOCIT
8=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
9=WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOCIT.COD_DOCIT < 990
10=AND DOC.COD_ESTAB = :COD_ESTAB AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00'AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
11=AND DOC.COD_AIDF >=0 AND DOC.PRINT_DOC = 'S' AND DOC.COD_MOD IN ('62') AND DOC.CANC_DOC <> 'S'
12=GROUP BY CST_ICMS, CFOP, ALIQ_ICMS,COD_DOC
