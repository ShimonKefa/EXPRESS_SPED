1  = SELECT 'D100' REG, DOC.COD_DOC, 
2  = CASE WHEN DOC.MOV_DOC = 'E' THEN 0 ELSE 1 END IND_OPER, CASE WHEN DOC.COD_AIDF = 0 THEN '1' ELSE '0' END IND_EMIT,
3  = CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.COD_PESS END COD_PART, CASE WHEN DOC.COD_MOD = '8' THEN '08' WHEN DOC.COD_MOD = '7' THEN '07' WHEN DOC.COD_MOD = '9' THEN '09' ELSE DOC.COD_MOD END COD_MOD, 
4  = CASE 
5  =   WHEN DOC.FINNFE_DOC = '2' THEN '06' -- DOC.FISCAL COMPLEMENTAR
6  =   WHEN DOC.TIPO_DOC = 'E' THEN '01'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
7  =   WHEN DOC.CANC_DOC = 'S' THEN '02' -- DOCUMENTO CANCELADO
8  =   WHEN DOC.TIPO_DOC = 'E' AND DOC.CANC_DOC = 'S' THEN '03'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
9  =   WHEN DOC.TIPO_DOC = 'E' AND DOC.FINNFE_DOC = '2' THEN '01'  -- ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR
10 =   WHEN DOC.STATUS_DOC = 'I' THEN '05'
11 =     ELSE    '00'  --DOCUMENTO REGULAR
12 =   END COD_SIT, 
13 =   DOC.SERIE_DOC SER, DOC.SUBSERIE_DOC SUB,
14 =   DOC.NUM_DOC, CASE WHEN DOC.COD_MOD = '57' THEN DOC.CHAVENFE_DOC ELSE '' END CHV_CTE, CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.EMISS_DOC END DT_DOC, CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.ENTSAID_DOC END DT_A_P, 
15 =   CASE WHEN DOC.CANC_DOC = 'S' THEN '' ELSE CASE WHEN DOC.COD_MOD = '57' THEN FINNFE_DOC ELSE '' END END TP_CT_e, CASE WHEN DOC.CANC_DOC = 'S' THEN '' ELSE CASE WHEN DOC.COD_MOD = '57' THEN CASE WHEN DOC.FINNFE_DOC <> '0' THEN DOC.CHAVENFE_DOC ELSE '' END END END CHV_CTE_REF, 
16 =   CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.TOTALG_DOC END VL_DOC,
20 =   CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE COALESCE(DOC.VRDESC_DOC,0) END VL_DESC, CASE WHEN DOC.CANC_DOC = 'S' THEN '' ELSE CASE WHEN DOC.MOV_DOC = 'S' THEN CASE WHEN DOC.MODFRETE_DOC IN ('1','3') THEN '1' ELSE '2' END ELSE '1' END END IND_FRT, 
21 =   CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE CASE WHEN DOC.COD_AIDF = 0 THEN DOC.TOTPRODBR_DOC ELSE DOC.FRETEAUTO_DOC END END VL_SERV,  CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.BCICMSE_DOC END VL_BC_ICMS, CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE DOC.ICMSE_DOC END VL_ICMS, CASE WHEN DOC.CANC_DOC = 'S' THEN NULL ELSE 0 END VL_NT, '' COD_INF, '' COD_CTA
23 =   FROM DOC 
24 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
25 = WHERE DOC.COD_ESTAB = :COD_ESTAB: AND 
26 = ((DOC.EMISS_DOC >= :DataHora_Inicial  AND DOC.EMISS_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'S') OR  (DOC.ENTSAID_DOC >= :DataHora_Inicial AND DOC.ENTSAID_DOC <= :DataHora_Final AND DOC.MOV_DOC = 'E'))
27 = AND DOC.COD_AIDF >= 0 AND DOC.ESP_DOC IN ('CT')
