1 = SELECT 'H005' REG, EXTRACT(MONTH FROM CAST(:PARAM_6 AS DATE)) MES, CAST(:PARAM_6 AS DATE) DT_INV , SUM(TOTALINV_PROD) VL_INV, '01' MOT_INV FROM INVENT 
2 = INNER JOIN PROD ON PROD.COD_ESTQ = INVENT.COD_ESTQ
3 = INNER JOIN GRP ON PROD.COD_GRP = GRP.COD_GRP
4 = WHERE (:PARAM_5 = 'S' AND DTINV_PROD = CAST(:PARAM_6 AS DATE)) AND NAT_ESTQ = 'F' AND COD_ESTAB = :COD_ESTAB AND INVENT.TOTALINV_PROD > 0 AND
5 = (SUBSTRING(GRP.CLTRIB_GRP FROM 1 FOR 5)  IN ('00.01', '01.00', '02.00', '03.00', '04.00', '05.01', '06.01', '10.00'))
6 = HAVING SUM(COALESCE(TOTALINV_PROD,0)) > 0
