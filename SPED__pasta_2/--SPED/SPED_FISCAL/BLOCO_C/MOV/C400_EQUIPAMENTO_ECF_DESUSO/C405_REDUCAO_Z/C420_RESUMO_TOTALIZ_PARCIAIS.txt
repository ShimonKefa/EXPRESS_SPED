[SQL]
1  = SELECT 'C420' REG, REDZITECF.SERIE ECF_FAB, REDZITECF.MOVIMENTO DATA, REDZECF.COOREDZ CRZ, IFIT_ECF.ALIQPAF COD_TOT_PAR, SUM(REDZITECF.TOTALLIQUIDO) VLR_ACUM_TOT, CASE WHEN SUBSTRING(IFIT_ECF.ALIQPAF FROM 1 FOR 1) = '0' THEN SUBSTRING(IFIT_ECF.ALIQPAF FROM 1 FOR 2) ELSE '' END NR_TOT, '' DESCR_NR_TOT FROM REDZITECF
2  = INNER JOIN REDZECF ON (REDZECF.COD_RZ = REDZITECF.COD_RZ) INNER JOIN IFIT_ECF ON (REDZITECF.ALIQ = IFIT_ECF.ALIQUOTA AND REDZITECF.SERIE = IFIT_ECF.SERIE) 
3  = WHERE REDZECF.COD_RZ = :COD_RZ: AND EXTRACT(DAY FROM REDZECF.MOVIMENTO) = :DIA:  AND REDZECF.SERIE = ':ECF_FAB:' AND REDZITECF.MOVIMENTO >= :Data_Inicial AND REDZITECF.MOVIMENTO <= :Data_Final
4  = GROUP BY IFIT_ECF.ALIQPAF, REDZITECF.SERIE, REDZITECF.MOVIMENTO, REDZECF.COOREDZ
5  = UNION ALL
6  = SELECT 'C420' REG, REDZECF.SERIE ECF_FAB, REDZECF.MOVIMENTO DATA, REDZECF.COOREDZ CRZ, 'DT' COD_TOT_PAR, COALESCE(REDZECF.DESCONTOS,0) VLR_ACUM_TOT, '' NR_TOT, '' DESCR_NR_TOT FROM REDZECF
7  = WHERE REDZECF.COD_RZ = :COD_RZ: AND EXTRACT(DAY FROM REDZECF.MOVIMENTO) = :DIA:  AND REDZECF.SERIE = ':ECF_FAB:' AND REDZECF.MOVIMENTO >= :Data_Inicial AND REDZECF.MOVIMENTO <= :Data_Final AND 
8  = COALESCE(REDZECF.DESCONTOS,0) > 0
9  = UNION ALL
10 = SELECT 'C420' REG, REDZECF.SERIE ECF_FAB, REDZECF.MOVIMENTO DATA, REDZECF.COOREDZ CRZ, 'AT' COD_TOT_PAR, COALESCE(REDZECF.ACRESCIMOS,0) VLR_ACUM_TOT, '' NR_TOT, '' DESCR_NR_TOT FROM REDZECF
11 = WHERE REDZECF.COD_RZ = :COD_RZ: AND EXTRACT(DAY FROM REDZECF.MOVIMENTO) = :DIA:  AND REDZECF.SERIE = ':ECF_FAB:' AND REDZECF.MOVIMENTO >= :Data_Inicial AND REDZECF.MOVIMENTO <= :Data_Final AND 
12 = COALESCE(REDZECF.ACRESCIMOS,0) > 0
13 = UNION ALL
14 = SELECT 'C420' REG, REDZECF.SERIE ECF_FAB, REDZECF.MOVIMENTO DATA, REDZECF.COOREDZ CRZ, 'Can-T' COD_TOT_PAR, COALESCE(REDZECF.CANCELAMENTOS,0) VLR_ACUM_TOT, '' NR_TOT, '' DESCR_NR_TOT FROM REDZECF
15 = WHERE REDZECF.COD_RZ = :COD_RZ: AND EXTRACT(DAY FROM REDZECF.MOVIMENTO) = :DIA:  AND REDZECF.SERIE = ':ECF_FAB:' AND REDZECF.MOVIMENTO >= :Data_Inicial AND REDZECF.MOVIMENTO <= :Data_Final AND 
16 = COALESCE(REDZECF.CANCELAMENTOS,0) > 0
