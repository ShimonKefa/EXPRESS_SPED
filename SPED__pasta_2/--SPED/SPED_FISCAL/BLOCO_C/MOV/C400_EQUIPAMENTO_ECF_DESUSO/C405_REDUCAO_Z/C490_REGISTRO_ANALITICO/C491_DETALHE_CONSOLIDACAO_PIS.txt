2 = CASE WHEN MAX(CST_PIS) IN (1,2,3) THEN MAX(ALIQ_PIS) ELSE 0 END ALIQ_PIS, '' QUANT_BC_PIS, '' ALIQ_PIS_QUANT,
3 = CASE WHEN MAX(CST_PIS) IN (1,2,3) THEN (SUM(VL_ITEM) * MAX(ALIQ_PIS)) / 100 ELSE 0 END VL_PIS, MAX(COD_CTA) COD_CTA
4 = FROM
5 = (
6 = SELECT 'C491' REG, CUPOMIT.COD_ESTAB||'-'||PROD.CODBAR_PROD COD_ITEM, 
7 = GRPIT.CST_PIS,

9 = CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
10= 0
11= ELSE
12= CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
13= ((CUPOMIT.TOTAL)) -
14= ((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_PIS,0) / 100)
15= ELSE
16= ((CUPOMIT.TOTAL)) -
17= ((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_PIS,0) / 100)
18= END
19= END VL_BC_PIS,
20= COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) ALIQ_PIS,
21= '' QUANT_BC_PIS, '' ALIQ_PIS_QUANT, 
22 = CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
23 = 0
24 = ELSE
25 = CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
26 = ((CUPOMIT.TOTAL)) -

28 = ELSE
29 = ((CUPOMIT.TOTAL)) -

33 = '' COD_CTA FROM CUPOMIT

36 = INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')

38 = INNER JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
39 = WHERE CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final AND CUPOMIT.COD_ESTAB = :COD_ESTAB:
40 = AND COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N')
41 = ) GROUP BY COD_ITEM, CFOP
