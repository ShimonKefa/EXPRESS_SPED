[SQL]
1  = SELECT 'C470' REG, CUPOMIT.DATA DATA, CUPOMIT.SERIE ECF_FAB, CUPOM.NUMERO NUM_DOC, PROD.CODBAR_PROD COD_ITEM, CUPOMIT.QTDE QTD, CASE WHEN CUPOMIT.CANCELADO = 'S' THEN CUPOMIT.QTDE ELSE 0 END QTD_CANC,
2  = PROD.UNSD_PROD UNID, CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0) VL_ITEM, CASE WHEN ALIQECF.ALIQUOTA = 'F' THEN '060' ELSE CASE WHEN ALIQECF.ALIQUOTA = 'I' THEN '040' ELSE CASE WHEN COALESCE(ALIQECF.PADRAONUMERICO,0) < 0.17 THEN '020' ELSE '000' END END END  CST_ICMS, 
3  = CASE WHEN ALIQECF.ALIQUOTA = 'F'  AND COALESCE(GRP.CLTRIB_GRP,'00') = '00' THEN '5405' WHEN ALIQECF.ALIQUOTA = 'F' AND GRP.CLTRIB_GRP  LIKE '07.01%' THEN '5656' ELSE '5102' END CFOP, COALESCE(CUPOMIT.ALIQICMS,0) * 100 ALIQ_ICMS,  
4  = NULL VL_PIS, 
5  = NULL VL_COFINS
6  = FROM CUPOMIT 
7  = INNER JOIN CUPOM ON (CUPOMIT.COD_CF = CUPOM.COD_CF)
8  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOMIT.COD_ESTAB)
9  = INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
10 = INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
11 = INNER JOIN ALIQECF ON (ALIQECF.ALIQUOTA = CUPOMIT.ALIQUOTA) 
12 = INNER JOIN GRP ON (GRP.COD_GRP = PROD.COD_GRP)
13 = INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
14 = WHERE CUPOMIT.SERIE = ':ECF_FAB:' AND EXTRACT(DAY FROM CUPOMIT.DATA) = :DIA: AND CUPOMIT.NUMERO = :NUM_DOC: AND CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final AND CUPOMIT.COD_ESTAB = :COD_ESTAB
15 = AND COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N')
