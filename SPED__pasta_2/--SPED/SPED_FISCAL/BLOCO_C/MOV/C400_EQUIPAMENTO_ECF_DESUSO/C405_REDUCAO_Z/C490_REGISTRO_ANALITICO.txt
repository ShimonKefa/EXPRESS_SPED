[SQL]
1 = SELECT REG, DATA, ECF_FAB, CST_ICMS, CFOP, ALIQ_ICMS, SUM(VL_OPR) VL_OPR, SUM(VL_BC_ICMS) VL_BC_ICMS, SUM(trunc(VL_ICMS,2)) VL_CMS, COD_OBS FROM (
2 = SELECT 'C490' REG, REDZITECF.MOVIMENTO DATA, REDZITECF.SERIE ECF_FAB, CASE WHEN ALIQECF.ALIQUOTA = 'F' THEN '060' ELSE CASE WHEN ALIQECF.ALIQUOTA = 'I' THEN '040' ELSE CASE WHEN COALESCE(ALIQECF.PADRAONUMERICO,0) < 0.17 THEN '020' ELSE '000' END END END  CST_ICMS,    
3 = CASE WHEN ALIQECF.ALIQUOTA = 'F' THEN '5405' ELSE '5102' END CFOP, (ALIQECF.PADRAONUMERICO * 100) ALIQ_ICMS, SUM(REDZITECF.TOTALLIQUIDO) VL_OPR, CASE WHEN ALIQECF.PADRAONUMERICO > 0 THEN SUM(REDZITECF.TOTALLIQUIDO) ELSE 0 END VL_BC_ICMS,
4 = SUM(COALESCE(REDZITECF.IMPOSTO,0)) VL_ICMS, '' COD_OBS FROM REDZITECF	
5 = INNER JOIN REDZECF ON (REDZITECF.COD_RZ = REDZECF.COD_RZ)
6 = LEFT JOIN ALIQECF ON (ALIQECF.ALIQUOTA = REDZITECF.ALIQ)
7 = WHERE REDZITECF.SERIE = ':ECF_FAB:' AND REDZITECF.COD_RZ = :COD_RZ: AND EXTRACT(DAY FROM REDZITECF.MOVIMENTO) = :DIA: AND REDZITECF.MOVIMENTO >= :Data_Inicial AND REDZITECF.MOVIMENTO <= :Data_Final
8 = GROUP BY ALIQECF.ALIQUOTA, REDZITECF.ALIQPAF, ALIQECF.PADRAONUMERICO, REDZITECF.MOVIMENTO, REDZITECF.SERIE 
9 = UNION ALL
10 = SELECT REG, DATA, ECF_FAB,  CST_ICMS, 
11 = CASE WHEN VEZES.LINHA = 1 THEN '5405' ELSE '5656' END CFOP, 
12 = ALIQ_ICMS, 
13 = CASE WHEN VEZES.LINHA = 1 THEN (-1) ELSE 1 END * VL_OPR, VL_BC_ICMS, VL_ICMS, COD_OBS FROM (
14 = SELECT 'C490' REG, CUPOM.DATA DATA, CUPOM.SERIE ECF_FAB, '060' CST_ICMS, '5656' CFOP, 0 ALIQ_ICMS, SUM(CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0)) VL_OPR, 0.0 VL_BC_ICMS, 0.0 VL_ICMS, '' COD_OBS FROM CUPOMIT
15 = INNER JOIN CUPOM ON (CUPOM.COD_CF = CUPOMIT.COD_CF)
16 = INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
17 = INNER JOIN GRP ON (GRP.COD_GRP = PROD.COD_GRP)
18 = WHERE COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N') AND (GRP.CLTRIB_GRP LIKE '07.01%') AND CUPOM.DATA >= :Data_Inicial AND CUPOM.DATA <= :Data_Final AND CUPOMIT.ALIQUOTA = 'F'
19 = GROUP BY CUPOM.SERIE, CUPOM.DATA
20 = ) MV,
21 = (SELECT 1 LINHA FROM RDB$DATABASE UNION ALL SELECT 2 LINHA FROM RDB$DATABASE) VEZES
22 = )GROUP BY REG, DATA, ECF_FAB, CST_ICMS, CFOP, ALIQ_ICMS, COD_OBS
