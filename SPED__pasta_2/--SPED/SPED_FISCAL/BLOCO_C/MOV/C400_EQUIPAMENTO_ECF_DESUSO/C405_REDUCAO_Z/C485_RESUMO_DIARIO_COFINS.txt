[SQL]

1  = SELECT 'C485' REG, ':ECF_FAB:' ECF_FAB, CAST(DATA AS DATE) DT_DOC, CST_COFINS, SUM(VL_ITEM) VL_ITEM, SUM(VL_BC_COFINS) VL_BC_COFINS,ALIQ_COFINS,MAX(QUANT_BC_COFINS) QUANT_BC_COFINS, MAX(ALIQ_COFINS_QUANT) ALIQ_COFINS_QUANT, 
2  = (SUM(VL_ITEM) * MAX(ALIQ_COFINS)) / 100 VL_COFINS, COD_ITEM, MAX(COD_CTA) FROM (
3  = SELECT 'C485' REG, GRPIT.CST_COFINS,CASE WHEN (COALESCE(CUPOMIT.DAITEM,0)) < 0 THEN (CUPOMIT.TOTAL - COALESCE(CUPOMIT.DAITEM,0)) ELSE (CUPOMIT.TOTAL) END VL_ITEM,
5  = CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
6  = 0
7  = ELSE
8  = CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
9  = ((CUPOMIT.TOTAL)) -
10 = ((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)
11 = ELSE
12 = ((CUPOMIT.TOTAL)) -
13 = ((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)
14 = END
15 = END VL_BC_COFINS,
16 = COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) ALIQ_COFINS,
17 = '' QUANT_BC_COFINS, '' ALIQ_COFINS_QUANT, 
18 = CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
19 = 0
20 = ELSE
21 = CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
22 = ((CUPOMIT.TOTAL)) -
23 = (((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)) * COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
24 = ELSE
25 = ((CUPOMIT.TOTAL)) -
26 = (((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)) * COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
27 = END
28 = END VL_COFINS,
29 = CUPOMIT.COD_ESTAB||'-'||PROD.CODBAR_PROD COD_ITEM, '' COD_CTA, CUPOMIT.DATA FROM CUPOMIT 
30 = INNER JOIN CUPOM ON (CUPOMIT.COD_CF = CUPOM.COD_CF)
31 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOMIT.COD_ESTAB)
32 = INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
33 = INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
34 = INNER JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
37 = WHERE CUPOMIT.DATA = ":DT_DIA:" AND CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final AND CUPOMIT.SERIE = ':ECF_FAB:' AND CUPOMIT.COD_ESTAB = :COD_ESTAB:
38 = AND COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N') 
39 = ) GROUP BY CST_COFINS, ALIQ_COFINS, COD_ITEM, DATA
