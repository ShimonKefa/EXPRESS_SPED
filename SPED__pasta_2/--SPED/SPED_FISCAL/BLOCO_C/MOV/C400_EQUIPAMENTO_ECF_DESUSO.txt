5=WHERE CUPOMIT.DATA >= :Data_Inicial || ' 00:00:00' AND CUPOMIT.DATA <= :Data_Final || ' 23:59:59' AND CUPOMIT.COD_ESTAB = :COD_ESTAB
6=) MV
