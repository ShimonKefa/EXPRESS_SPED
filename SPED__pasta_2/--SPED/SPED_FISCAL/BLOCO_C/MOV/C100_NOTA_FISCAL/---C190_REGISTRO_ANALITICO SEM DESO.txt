[SQL]
1 = SELECT 'C190' REG, MV.COD_DOC, MV.CST_ICMS, MV.CFOP, MV.ALIQ_ICMS ALIQ_ICMS, COALESCE(SUM(MV.VL_ITEM - MV.VL_DESC + COALESCE(VL_IPI,0)+ COALESCE(MV.VL_OUT_DA,0)+COALESCE(MV.OUTROS_VL,0)),0) VL_OPR, SUM(MV.VL_BC_ICMS) VL_BC_ICMS, SUM(MV.VL_ICMS) VL_ICMS, 
2 = SUM(MV.VL_BC_ICMS_ST) VL_BC_ICMS_ST, SUM(MV.VL_ICMS_ST) VL_ICMS_ST, 
3 = CASE WHEN MV.CST_ICMS LIKE '%20' OR MV.CST_ICMS LIKE '%70' THEN SUM(MV.VL_ITEM - MV.VL_BC_ICMS) ELSE 0 END VL_RED_BC, COALESCE(SUM(MV.VL_IPI),0) VL_IPI, '' COD_OBS 
4 = FROM (:SQL_DOCITS:) MV
5 = WHERE MV.COD_MOD IN ('01','55','65','4', '04') AND COALESCE(MV.CANC_DOCIT,'N')  = 'N' AND COALESCE(MV.DEL_DOCIT,'N') = 'N'
6 = GROUP BY REG, COD_DOC, CST_ICMS, CFOP, ALIQ_ICMS, CST_ICMS, COD_OBS
