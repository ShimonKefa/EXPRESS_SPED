2 = DOC.EMISS_DOC DT_PGTO FROM DOCIT
3 = INNER JOIN DOC ON DOC.COD_DOC = DOCIT.COD_DOC
4 = WHERE ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR 
5 = (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E')) AND
6 = DOCIT.COD_DOCIT <= 990 AND DOC.PRINT_DOC = 'S' AND DOC.COD_MOD = '55' AND COALESCE(DOCIT.DALIQICMS_DOCIT,0) > 0 AND DOC.COD_AIDF > 0 
7 = AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.CANC_DOC = 'N' AND DOC.COD_DOC = :COD_DOC:
8 = GROUP BY COD_DOC, REG, COD_DA,  DOC.UF_PESS, DOCIT.DALIQICMS_DOCIT, DOC.EMISS_DOC, DOC.EMISS_DOC, DOC.NUM_DOC
9 = UNION ALL
10 = SELECT DOC.COD_DOC, 'C112' REG, 0 COD_DA,  'ES' UF, DOC.OBSLIV_DOC NUM_DA, DOC.EXTERMPG_DOC COD_AUT, SUM(DOC.ICMSAUTO_DOC) VL_DA, DOC.EMISS_DOC DT_VCTO, 
11 = COALESCE(DOC.DTVENCFR_DOC,DOC.EMISS_DOC) DT_PGTO FROM DOC
12 = WHERE ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR 
13 = (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E')) AND
14 = COAlesCe(DOC.CANc_DOC,'N') <> 'S'AND DOC.PRINT_DOC = 'S' AND DOC.COD_MOD = '55' AND 
15 = COALESCE(DOC.ICMSAUTO_DOC,0) > 0 AND DOC.COD_AIDF > 0  AND (DOC.EXTERMPG_DOC <> '' OR DOC.OBSLIV_DOC <> '')
16 = AND DOC.COD_ESTAB = :COD_ESTAB AND DOC.CANC_DOC = 'N' AND DOC.COD_DOC = :COD_DOC: 
17 = GROUP BY COD_DOC, REG, COD_DA,  UF_PESS, DOC.DTVENCFR_DOC, DOC.EMISS_DOC, DOC.OBSLIV_DOC, DOC.EXTERMPG_DOC
