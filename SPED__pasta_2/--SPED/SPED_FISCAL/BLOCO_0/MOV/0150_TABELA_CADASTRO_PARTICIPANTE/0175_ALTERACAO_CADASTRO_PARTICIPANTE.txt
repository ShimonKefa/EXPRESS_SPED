[SQL]
1=SELECT DISTINCT '0175' REG, DT_ALT, NR_CAMPO, MIN(CONT_ANT) CONT_ANT, COD_PART FROM
2=(
3=SELECT CAST(CAST(DHALT_LOG AS CHAR(11 )) AS DATE ) DT_ALT,
4=CASE
5=WHEN CAMPO_LOG = 'NOM_PESS' THEN 03
6=WHEN CAMPO_LOG = 'END_PESS' THEN 10
7=WHEN CAMPO_LOG = 'NUM_PESS' THEN 11
8=WHEN CAMPO_LOG = 'COMPL_PESS' THEN 12
9=WHEN CAMPO_LOG = 'BAIRRO_PESS' THEN 13
10=END NR_CAMPO, VRANT_LOG CONT_ANT, CHAVEINT1_LOG COD_PART
11=FROM LOG_CAD
12= INNER JOIN (:SQL_0150:) P ON P.COD_PART = LOG_CAD.CHAVEINT1_LOG
13=WHERE TABELA_LOG = 'PESSOAS' AND (DHALT_LOG >= CAST(:DataHora_Inicial AS TIMESTAMP)) AND (DHALT_LOG <= CAST(:DataHora_Final AS TIMESTAMP)) AND (VRANT_LOG <> '')
14=)MV
15=WHERE MV.NR_CAMPO IN (03,04,05,06,08,09,10,11,12,13) AND MV.COD_PART = :COD_PART: 
16=GROUP BY REG, DT_ALT, NR_CAMPO, COD_PART
