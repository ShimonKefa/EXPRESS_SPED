1 = SELECT '0190' REG, UNID, DESCR FROM (
2 = --ITENS DAS NOTAS FISCAS
3 = SELECT UNID, COALESCE(MAX(DESCR), UNID || '.') DESCR FROM (
4 = SELECT CASE  WHEN PROD.UNSD_PROD = 'M³' THEN 'M3' WHEN PROD.UNSD_PROD = 'M²' THEN 'M2' WHEN PROD.UNSD_PROD = 'PÇ' THEN 'PC' ELSE  UPPER(PROD.UNSD_PROD) END UNID,
5 = UPPER(UNID.NOM_UN) DESCR FROM PROD 
6 = INNER JOIN (:SQL_DOCITS) MV ON (MV.TPEMI_DOC = 'T' AND PROD.COD_ESTQ = MV.COD_ESTQ)
7 = LEFT JOIN UNID ON (UNID.COD_UN = PROD.UNSD_PROD)
8 = WHERE (PROD.COD_ESTAB = :COD_ESTAB) AND (MV.MOV = 'E' AND MV.COD_MOD NOT IN ('06','57','65','21', '22', '08','NFSE', '66')) AND PROD.UNSD_PROD IS NOT NULL
9 = UNION ALL
10 = --REGISTRO H010 E K200
11 = SELECT CASE WHEN PROD.UNSD_PROD = 'M³' THEN 'M3' WHEN PROD.UNSD_PROD = 'M²' THEN 'M2' WHEN PROD.UNSD_PROD = 'PÇ' THEN 'PC' ELSE  UPPER(PROD.UNSD_PROD) END UNID,
12 = UPPER(UNID.NOM_UN) DESCR FROM PROD 
13 = INNER JOIN (:SQL_INVENT) MV ON (PROD.COD_ESTQ = MV.COD_ESTQ)
14 = LEFT JOIN UNID ON (UNID.COD_UN = PROD.UNSD_PROD)
15 = WHERE (PROD.COD_ESTAB = :COD_ESTAB) 
16 = ) GROUP BY UNID
17 = )
