[SQL]
1=SELECT "K255" REG, DOC.EMISS_DOC DT_CONS,
2=PROD.CODBAR_PROD COD_ITEM, SUM(DOCIT.QTDEESTQ_DOCIT) QTD, '' COD_INS_SUBST
3=FROM DOCIT
4=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
5=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
6=INNER JOIN 
7=(
8=SELECT DISTINCT COD_ESTQ, DOCIT.CODREF_DOC, DOCIT.CODREF_DOCIT
9=FROM DOCIT
10=INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
11=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
12=WHERE CFOP.ESTAT_CFOP = "RET" AND DOC.EMISS_DOC >= CAST(:Data_Inicial AS DATE) AND DOC.EMISS_DOC <= CAST(:Data_Final AS DATE)
13=AND DOC.CANC_DOC = "N" AND DOC.PRINT_DOC = "S" AND DOC.MOV_DOC = "E"
14=) DOC_RET ON (DOC_RET.CODREF_DOC = DOC.COD_DOC AND DOC_RET.CODREF_DOCIT = DOCIT.COD_DOCIT)
15=GROUP BY REG, DT_CONS, COD_ITEM
