1=SELECT "K230" REG, MOV.COD_MV,  MOV.DTINI_MV DT_INI_OP, MOV.DTFIM_MV DT_FIN_OP, MOV.NUM_MV COD_DOC_OP, PROD.CODBAR_PROD COD_ITEM,

3=FROM MOVIT
4=INNER JOIN PROD ON (PROD.COD_ESTQ = MOVIT.COD_ESTQ)
5=INNER JOIN MOV ON (MOV.COD_MV = MOVIT.COD_MV)
6=WHERE MOVIT.MOV_MVIT = 'E' AND ((MOV.DTINI_MV >= CAST(:Data_Inicial AS DATE) AND MOV.DTINI_MV <= CAST(:Data_Final AS DATE)) OR (MOV.DTFIM_MV >= CAST(:Data_Inicial AS DATE) AND MOV.DTFIM_MV <= CAST(:Data_Final AS DATE)))
7=AND ':IND_MOV:' = 0 AND CAST(:Data_Inicial AS DATE) >= '01/01/2017' AND MOV.COD_ESTAB = :COD_ESTAB:
