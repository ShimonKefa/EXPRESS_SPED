CREATE OR ALTER VIEW VSPED_DOCIT(
    REG,
    NUM_DOC,
    DT_DOC,
    DT_E_S,
    NUM_ITEM,
    COD_ITEM,
    DESCR_COMPL,
    QTDE,
    UN,
    VL_ITEM_FISCAL,
    VL_ITEM_CONTRIB,
    VL_DESC,
    IND_MOV,
    CST_ICMS,
    CFOP,
    INTCFOP,
    COD_NAT,
    VL_BC_ICMS,
    ALIQ_ICMS,
    VL_ICMS,
    VL_BC_ICMS_ST,
    ALIQ_ST,
    VL_ICMS_ST,
    CST_IPI,
    COD_ENQ,
    VL_BC_IPI,
    ALIQ_IPI,
    VL_IPI,
    CST_PIS,
    VL_BC_PIS,
    ALIQ_PIS,
    VL_PIS,
    CST_COFINS,
    VL_BC_COFINS,
    ALIQ_COFINS,
    VL_COFINS,
    VL_ISS,
    DESC_CRED,
    COD_CTA,
    NAT_BC_CRED,
    COD_SIT,
    COD_ESTQ,
    EX_IPI,
    SER,
    TOTALDV_DOCIT,
    TIPO_TRIB,
    COD_NCM,
    FIN_CFOP,
    TP_ESC,
    ALIQPIS_ESTB,
    ALIQCOFINS_ESTB,
    RAIZ_CNPJ,
    SPED_PIS_COFINS,
    ESTAT_CFOP,
    IND_APUR,
    UF,
    DEL_DOCIT,
    VL_OUT_DA,
    CANC_DOCIT,
    CANC_DOC,
    CNPJCTB,
    OUTROS_VL,
    IND_EMIT,
    IND_OPER,
    COD_AIDF,
    COD_MOD,
    MOV_DOC,
    TPEMI_DOC,
    CSNAEFD_ESTB,
    RECBRUTACS_ESTB,
    NAT_REC,
    COD_DOC,
    COD_DOCIT,
    COD_ESTAB,
    ICMS_SN,
    DIFAL,
    ESP_DOC,
	PRINT_DOC)
AS
SELECT
"DOCITS" REG, 
VSPED.NUM_DOC, 
VSPED.DT_DOC,  VSPED.DT_E_S,
IIF(COALESCE(DOCIT.ORDEM_DOCIT,0) = 0, DOCIT.COD_DOCIT, DOCIT.ORDEM_DOCIT) NUM_ITEM, 
VSPED.COD_ITEM, 
VSPED.DESCR_COMPL, 
IIF(COALESCE(DOCIT.QTDEESTQ_DOCIT,0) = 0, NULL, DOCIT.QTDEESTQ_DOCIT) QTDE,
VSPED.UN,
(DOCIT.TOTAL_DOCIT + IIF((VSPED.FINNFE_DOC = "2" AND DOCIT.TOTAL_DOCIT = 0) OR (VSPED.CANC_DOC = "S") OR (VSPED.STATUS_DOC = "I"), 0, COALESCE(DOCIT.VRDESC_DOCIT,0) + COALESCE(DOCIT.IPI_DOCIT,0) +
IIF(COALESCE(DOCIT.ADDICMSR_DOCIT,"N") = "S", COALESCE(DOCIT.ICMSR_DOCIT,0) - COALESCE(DOCIT.ICMSRE_DOCIT,0),    0) - IIF(COALESCE(VSPED.ATIV2_ESTB,"0") = "0", COALESCE(DOCIT.IPIE_DOCIT,0), 0))) VL_ITEM_FISCAL,
DOCIT.VLITEM_DOCIT VL_ITEM_CONTRIB, 
COALESCE(DOCIT.VRDESC_DOCIT,0) VL_DESC, 
VSPED.IND_MOV,
RIGHT("000" || COALESCE(DOCIT.CSTICMSE_DOCIT,999),3) CST_ICMS,
DOCIT.CFOP_DOCIT CFOP, DOCIT.INTCFOP_DOCIT INTCFOP,
VSPED.COD_NAT, 
IIF(VSPED.CODBAR3_PROD IN ("A28793","C28874"), 0, COALESCE(DOCIT.BCICMSE_DOCIT,0.0)) VL_BC_ICMS,
IIF(VSPED.CODBAR3_PROD IN ("A28793","C28874"), 0, COALESCE(DOCIT.ALIQICMSE_DOCIT,0)) ALIQ_ICMS, IIF(VSPED.CODBAR3_PROD IN ("A28793","C28874"), 0, COALESCE(DOCIT.ICMSE_DOCIT,0.0)) VL_ICMS, COALESCE(DOCIT.BCICMSRE_DOCIT,0) VL_BC_ICMS_ST,
COALESCE(DOCIT.ALIQICMSR_DOCIT,0) ALIQ_ST, COALESCE(DOCIT.ICMSRE_DOCIT,0) VL_ICMS_ST,
IIF(COALESCE(VSPED.ATIV2_ESTB,"0") <> "0", NULL, RIGHT("00" || COALESCE(DOCIT.CSTIPIE_DOCIT,0),2)) CST_IPI, "999" COD_ENQ, IIF(COALESCE(VSPED.ATIV2_ESTB,"0") <> "0", NULL, COALESCE(DOCIT.BCIPIE_DOCIT,0)) VL_BC_IPI,
IIF(COALESCE(VSPED.ATIV2_ESTB,"0") <> "0", NULL, COALESCE(DOCIT.ALIQIPIE_DOCIT,0)) ALIQ_IPI, IIF(COALESCE(VSPED.ATIV2_ESTB,"0") <> "0", NULL, COALESCE(DOCIT.IPIE_DOCIT,0)) VL_IPI,

DOCIT.CSTPISE_DOCIT CST_PIS, DOCIT.BCPISE_DOCIT VL_BC_PIS, DOCIT.ALIQPISE_DOCIT ALIQ_PIS, DOCIT.PISE_DOCIT VL_PIS, DOCIT.CSTCOFINSE_DOCIT CST_COFINS, DOCIT.BCCOFINSE_DOCIT VL_BC_COFINS, DOCIT.ALIQCOFINSE_DOCIT ALIQ_COFINS, DOCIT.COFINSE_DOCIT VL_COFINS,
DOCIT.ISS_DOCIT VL_ISS, VSPED.DESC_CRED,
VSPED.COD_CTA, VSPED.NAT_BC_CRED,
VSPED.COD_SIT,
DOCIT.COD_ESTQ, VSPED.EX_IPI, VSPED.SER, IIF(COALESCE(DOCIT.CODREF_DOC,0) = 0 AND COALESCE(DOCIT.CODREF_DOCIT,0) = 0, 0, COALESCE(DOCIT.TOTALDV_DOCIT,0)) TOTALDV_DOCIT, VSPED.TIPO_TRIB, VSPED.COD_NCM,
VSPED.FIN_CFOP, VSPED.TP_ESC, VSPED.ALIQPIS_ESTB, VSPED.ALIQCOFINS_ESTB, VSPED.RAIZ_CNPJ, VSPED.SPED_PIS_COFINS, VSPED.ESTAT_CFOP, "0" IND_APUR, VSPED.UF, DOCIT.DEL_DOCIT, DOCIT.OUTDESP_DOCIT VL_OUT_DA, DOCIT.CANC_DOCIT,

VSPED.CANC_DOC, NULL CNPJCTB, COALESCE(DOCIT.FRETE_DOCIT,0) + COALESCE(DOCIT.SEGURO_DOCIT,0) OUTROS_VL, VSPED.IND_EMIT, VSPED.IND_OPER, VSPED.COD_AIDF, VSPED.COD_MOD, VSPED.MOV_DOC, VSPED.TPEMI_DOC,  VSPED.CSNAEFD_ESTB, VSPED.RECBRUTACS_ESTB, VSPED.NAT_REC, VSPED.COD_DOC, VSPED.COD_DOCIT, VSPED.COD_ESTAB,
DOCIT.ICMSA_DOCIT ICMS_SN, DOCIT.DALIQICMS_DOCIT DIFAL, VSPED.ESP_DOC, VSPED.PRINT_DOC
FROM DOCIT
INNER JOIN VSPED ON VSPED.COD_DOC = DOCIT.COD_DOC --AND VSPED.COD_DOCIT = DOCIT.COD_DOCIT
;
