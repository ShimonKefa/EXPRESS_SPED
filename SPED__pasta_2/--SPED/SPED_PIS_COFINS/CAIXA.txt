[SQL]
1  = SELECT "CAIXA" REG, FATURBX.DTPAGTO_FAT DT_DOC, 
2  = TRUNC((:SQL_VL_ITEM:) * FATURBX.INDICE,2) VL_REC_CAIXA,
3  = CASE WHEN COALESCE(DOC.ESCCS_DOC,'N') IN ('S','P','M') THEN COALESCE(DOCIT.CSTPISE_DOCIT,GRPIT.CST_PIS) ELSE GRPIT.CST_PIS END CST_PIS,
4  = TRUNC(FATURBX.DESC_FAT,2) VL_DESC_PIS,
5  = TRUNC((:SQL_VL_BC_PIS_COFINS:) * FATURBX.INDICE, 2) VL_BC_PIS,
6  = COALESCE(CASE WHEN COALESCE(DOC.ESCCS_DOC,'N') IN ('S','P','M') THEN COALESCE(DOCIT.ALIQPISE_DOCIT,GRPIT.ALIQ_PIS) ELSE GRPIT.ALIQ_PIS END,0) ALIQ_PIS,
7  = TRUNC(IIF(COALESCE(DOC.ESCCS_DOC,'N') IN ('S','P','M'), DOCIT.PISE_DOCIT * FATURBX.INDICE, (:SQL_VL_BC_PIS_COFINS:) * FATURBX.INDICE * (COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) / 100)), 2) VL_PIS,
8  = CASE WHEN COALESCE(DOC.ESCCS_DOC,'N') IN ('S','C','M') THEN COALESCE(DOCIT.CSTCOFINSE_DOCIT,GRPIT.CST_COFINS) ELSE GRPIT.CST_COFINS END CST_COFINS,
9  = TRUNC(FATURBX.DESC_FAT, 2) VL_DESC_COFINS,
10 = TRUNC((:SQL_VL_BC_PIS_COFINS:) * FATURBX.INDICE, 2) VL_BC_COFINS,
11 = COALESCE(CASE WHEN COALESCE(DOC.ESCCS_DOC,'N') IN ('S','C','M') THEN COALESCE(DOCIT.ALIQCOFINSE_DOCIT,GRPIT.ALIQ_COFINS) ELSE GRPIT.ALIQ_COFINS END,0) ALIQ_COFINS,
12 = TRUNC(IIF(COALESCE(DOC.ESCCS_DOC,'N') IN ('S','P','M'), DOCIT.COFINSE_DOCIT * FATURBX.INDICE, (:SQL_VL_BC_PIS_COFINS:) * FATURBX.INDICE * (COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100)), 2) VL_COFINS,
13 = CASE WHEN RIGHT('00' || DOC.COD_MOD,2) = 'SE' THEN '98' ELSE RIGHT('00' || DOC.COD_MOD,2)  END COD_MOD,
14 = DOCIT.CFOP_DOCIT CFOP,
15 = CASE WHEN CHAR_LENGTH(CFOP.COD_CTA) > 0 THEN CFOP.COD_CTA ELSE GRPIT.CODCTB_CTA END COD_CTA,
16 = FATURBX.COD_MOD || ' - ' || FATURBX.NUM_FAT INFO_COMPL,
17 = DOC.COD_ESTAB,
18 = COALESCE(PROD.NATRECCS_PROD, GRPIT.NATREC_TRIB) NAT_REC,
19 = ESTAB.CNPJCPF_ESTB CNPJ_CPF,
20 = CASE WHEN COALESCE(FATURBX.CODF_PESS,0) <> 0 THEN 02 ELSE 01 END IND_REC,
21 = DOC.NUM_DOC,
22 = DOC.NOM_PESS,
23 = SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) RAIZ_CNPJ,
24 = ESTAB.CNPJCPF_ESTB CNPJ_ESTAB, ESTAB.ALIQPIS_ESTB, ESTAB.ALIQCOFINS_ESTB,
25 = IIF(DOC.MOV_DOC = 'E', 0, 1) IND_OPER, CFOP.ESTAT_CFOP ESTAT_CFOP, 
26 = DOC.COD_FAT, FATURBX.INDICE, FATURBX.VRPAGO_FAT, ESTAB.CNPJCPF_ESTB ESTAB_CNPJ
27 = FROM DOCIT 
28 = INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
29 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
30 = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
31 = INNER JOIN CFOP ON (CFOP.COD_CFOP = RIGHT( '0000' || DOCIT.CFOP_DOCIT,4) AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
32 = INNER JOIN GRP ON (GRP.COD_GRP = PROD.COD_GRP)
33 = INNER JOIN GRPIT ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP) = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
34 = INNER JOIN 
35 = (
36 = 	SELECT MAX(DTPAGTO_FAT) DTPAGTO_FAT, MAX(FATURBX.COD_MOD) COD_MOD, LIST(FATURBX.NUM_FAT) NUM_FAT, FATURBX.COD_FAT COD_FAT, COALESCE(MAX(COBRANCA.CODF_PESS),0) CODF_PESS, 
37 = 	SUM(FATURBX.DESC_FAT) DESC_FAT,
38 =	IIF(MAX(DOC.TOTFIN_DOC) <> 0, SUM(FATURBX.SALDO_FAT) / MAX(DOC.TOTFIN_DOC), 0) INDICE, SUM(FATURBX.VRPAGO_FAT) VRPAGO_FAT FROM FATURBX
39 =	INNER JOIN DOC ON (DOC.COD_FAT = FATURBX.COD_FAT)
40 =	LEFT JOIN COBRANCA ON (COBRANCA.COD_COBR = FATURBX.COD_COBR)
41 =	WHERE FATURBX.DTPAGTO_FAT >= :Data_Inicial || ' 00:00:00' AND FATURBX.DTPAGTO_FAT <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = "S"
42 =	GROUP BY FATURBX.COD_FAT
43 = ) FATURBX ON (FATURBX.COD_FAT = DOC.COD_FAT)
44 = LEFT JOIN MOD ON (MOD.COD_MOD = DOC.COD_MOD)
45 = LEFT JOIN  (SELECT CODREF_DOC, CODREF_DOCIT, TOTAL_DOCIT + COALESCE(FRETE_DOCIT,0) + COALESCE(SEGURO_DOCIT,0) + COALESCE(OUTDESP_DOCIT,0) + COALESCE(IPI_DOCIT,0) TOTALDV_DOCIT FROM DOCIT) DOCITDV ON (DOCITDV.CODREF_DOC = DOCIT.COD_DOC AND DOCITDV.CODREF_DOCIT = DOCIT.COD_DOCIT)
46 = WHERE 
47 = DOCIT.COD_DOCIT < 990 AND DOC.COD_AIDF >=0 AND DOC.PRINT_DOC = "S" AND COALESCE(GRPIT.SPEDPISCOFINS_TRIB,'N') = 'S' AND
48 = DOC.CANC_DOC <> "S" AND COALESCE(DOCIT.CANC_DOCIT,"N") <> "S" AND COALESCE(DOCIT.DEL_DOCIT,"N") <> "S" AND ((DOC.MOV_DOC = 'S' AND ESTAB.INCTRIB1_ESTB = '2') OR (DOC.MOV_DOC IN ("S",'E') AND ESTAB.INCTRIB1_ESTB <> '2')) 
49 = AND (COALESCE(MOD.ESP_MOD,"NF") IN ('NFSE','ISS','NF') OR (DOC.MOV_DOC = "E" AND COALESCE(MOD.ESP_MOD,"NF") IN ("CT") AND DOC.COD_AIDF = 0))  
50 = AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ'
