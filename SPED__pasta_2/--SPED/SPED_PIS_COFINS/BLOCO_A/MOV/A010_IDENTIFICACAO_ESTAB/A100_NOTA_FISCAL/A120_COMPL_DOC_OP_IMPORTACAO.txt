1 = SELECT DISTINCT 'A120' REG, DOC.COD_DOC COD_DOC,

4 = ' ' NUM_ACDRAW, GRPIT.CLCTB_CFOP
5 = FROM DOCIT2 INNER JOIN DOC ON (DOC.COD_DOC = DOCIT2.COD_DOC)
6 = INNER JOIN DOCIT ON (DOCIT.COD_DOC = DOCIT2.COD_DOC AND DOCIT.COD_DOCIT = DOCIT2.COD_DOCIT)
7 = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
8 = INNER JOIN PROD ON PROD.COD_ESTQ = DOCIT.COD_ESTQ
9 = INNER JOIN GRP ON GRP.COD_GRP = PROD.COD_GRP
10 = INNER JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP AND GRPIT.COD_ESTAB = :COD_ESTAB)
11 = WHERE DOC.MOV_DOC = 'E' AND DOC.UF_PESS = 'EX' AND DOC.COD_AIDF >= 0 AND GRPIT.SPEDPISCOFINS_TRIB = 'S' AND ':COD_INC_TRIB:' = '1'
12 = AND DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.COD_ESTAB = :COD_ESTAB
