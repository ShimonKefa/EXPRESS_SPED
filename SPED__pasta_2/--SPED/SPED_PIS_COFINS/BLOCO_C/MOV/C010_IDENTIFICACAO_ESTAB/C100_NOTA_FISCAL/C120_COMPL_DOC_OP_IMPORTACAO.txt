[SQL]
1  = SELECT DISTINCT 'C120' REG, DOC.COD_DOC,
2  = 0 COD_DOC_IMP, CAST(DOCIT2.NUMDOC_DOCIT2 || '-' || DOCIT.COD_DOCIT AS VARCHAR(32000)), 
3  = COALESCE(DOCIT.PIS_DOCIT,0) VR_PIS_IMP, COALESCE(DOCIT.COFINS_DOCIT,0) VR_COFINS_IMP,
4  = ' ' NUM_ACDRAW
5  =  FROM DOCIT2 INNER JOIN DOC ON (DOC.COD_DOC = DOCIT2.COD_DOC)
6  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
7  =  INNER JOIN DOCIT ON (DOCIT.COD_DOC = DOCIT2.COD_DOC AND DOCIT.COD_DOCIT = DOCIT2.COD_DOCIT)
8  = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
9  = INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
10 = INNER JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
11  = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' AND DOC.COD_DOC = :COD_DOC: AND DOC.MOV_DOC = 'E' AND DOC.UF_PESS = 'EX' AND :COD_INC_TRIB: = '1'
12  = AND DOC.COD_AIDF >= 0 
13  = AND DOC.COD_ESTAB = :COD_ESTAB: AND ((DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') OR (DOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'E'))
14 = AND COALESCE(GRPIT.SPEDPISCOFINS_TRIB,'N') = 'S' --:COD_SIT: NOT IN ('02', '03', '04', '05')
