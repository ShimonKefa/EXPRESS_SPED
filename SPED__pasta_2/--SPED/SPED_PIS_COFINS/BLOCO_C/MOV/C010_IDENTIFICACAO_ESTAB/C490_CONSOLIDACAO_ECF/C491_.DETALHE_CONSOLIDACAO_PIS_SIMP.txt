[SQL]
1=SELECT MAX(REG) REG, COD_ITEM,  ':RAIZ_CNPJ:' RAIZ_CNPJ, MAX(CST_PIS) CST_PIS, CFOP, SUM(VL_ITEM) VL_ITEM, SUM(VL_BC_PIS) VL_BC_PIS,
2=CASE WHEN MAX(CST_PIS) IN (1,2,3) THEN MAX(ALIQ_PIS) ELSE 0 END ALIQ_PIS, '' QUANT_BC_PIS, '' ALIQ_PIS_QUANT,
3=SUM(TRUNC(VL_PIS,5)) VL_PIS, MAX(COD_CTA) COD_CTA, MAX(VENDA) VENDA, MAX(NAT_REC) NAT_REC, COD_ESTAB COD_ESTAB, :IND_ESCRI: IND_ESCRI
4=FROM
5=(
6=SELECT 'C491' REG, PROD.CODBAR_PROD COD_ITEM,
7=GRPIT.CST_PIS,
8=CASE WHEN PROD.ALIQECF_PROD = 'F' THEN '5405' ELSE '5102' END CFOP, CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0) VL_ITEM,
9=CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
10=0
11=ELSE
12=CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
13=((CUPOMIT.TOTAL)) -
14=((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_PIS,0) / 100)
15=ELSE
16=((CUPOMIT.TOTAL)) -
17=((CUPOMIT.TOTAL) * COALESCE(GRPIT.RDBC_PIS,0) / 100)
18=END
19=END VL_BC_PIS,
20=COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) ALIQ_PIS,
21='' QUANT_BC_PIS, '' ALIQ_PIS_QUANT,
22=CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
23=0
24=ELSE
25=(CUPOMIT.TOTAL + COALESCE(CUPOMIT.DACUPOM,0) - (CUPOMIT.TOTAL * COALESCE(GRPIT.RDBC_PIS,0) / 100))  * COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) / 100
26=END VL_PIS,
27=COALESCE(CFOP.CODCTB_CTA, GRPIT.CODCTB_CTA) COD_CTA, 'S' VENDA, COALESCE(PROD.NATRECCS_PROD, GRPIT.NATREC_TRIB) NAT_REC, CUPOMIT.COD_ESTAB COD_ESTAB FROM CUPOMIT
28=INNER JOIN CUPOM ON (CUPOMIT.COD_CF = CUPOM.COD_CF)
29=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOMIT.COD_ESTAB)
30=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
31=INNER JOIN PROD ON (PROD.COD_ESTQ = CUPOMIT.COD_ESTQ)
32=INNER JOIN GRPIT ON (PROD.CODSC_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
33=WHERE CUPOMIT.DATA >= :Data_Inicial AND CUPOMIT.DATA <= :Data_Final
34=AND COALESCE(CUPOMIT.COD_ESTQ,0) <> 0  AND (CUPOMIT.CANCELADO = 'N' AND CUPOM.CANCELADO='N')
35=) GROUP BY COD_ITEM, CFOP, COD_ESTAB
