3=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
4=INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
