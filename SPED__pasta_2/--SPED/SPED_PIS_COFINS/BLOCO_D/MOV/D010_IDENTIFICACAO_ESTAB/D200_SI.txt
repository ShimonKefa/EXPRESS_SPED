[SQL]
1=SELECT DISTINCT DOC.COD_DOC, 'D200' REG, GRPIT.CST_PIS CST_PIS, (DOC.TOTALG_DOC) VL_ITEM,
2=CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
3=0
4=ELSE
5=(DOC.TOTALG_DOC) + ((DOC.TOTALG_DOC) * COALESCE(GRPIT.RDBC_PIS,0) / 100)
6=END VL_BC_PIS,
7=COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) ALIQ_PIS,
8=CASE WHEN COALESCE(GRPIT.BC_PIS,'S') = 'N' THEN
9=0
10=ELSE
11=((DOC.TOTALG_DOC) - ((DOC.TOTALG_DOC) * COALESCE(GRPIT.RDBC_PIS,0) / 100)) * COALESCE(GRPIT.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) / 100
12=END VL_PIS,
13=GRPIT.CST_COFINS CST_COFINS,
14=CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
15=0
16=ELSE
17=(DOC.TOTALG_DOC) + ((DOC.TOTALG_DOC) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)
18=END VL_BC_COFINS,
19=COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) ALIQ_COFINS,
20=CASE WHEN COALESCE(GRPIT.BC_COFINS,'S') = 'N' THEN
21=0
22=ELSE
23=((DOC.TOTALG_DOC) - ((DOC.TOTALG_DOC) * COALESCE(GRPIT.RDBC_COFINS,0) / 100)) * COALESCE(GRPIT.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
24=END VL_COFINS,
25=COALESCE(CFOP.CODCTB_CTA, GRPIT.CODCTB_CTA) COD_CTA, GRPIT.NATREC_TRIB NAT_REC, SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) RAIZ_CNPJ, 'S' VENDA
26=FROM DOC
27=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) INNER JOIN DOCIT ON DOCIT.COD_DOC = DOC.COD_DOC
28=INNER JOIN CFOP ON (CAST(CFOP.COD_CFOP AS VARCHAR(4)) =  DOC.CFOPS_DOC  AND CFOP.INT_CFOP = COALESCE(DOC.INTCFOP_DOC,'00'))
29=INNER JOIN GRPIT ON (GRPIT.COD_GRP = 999 AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
30=WHERE DOC.EMISS_DOC >= :DataHora_Inicial AND DOC.EMISS_DOC <= :DataHora_Final AND DOC.COD_AIDF >= 0
31=AND DOC.ESP_DOC IN ('CT') AND DOC.MOV_DOC = 'S' AND ':IND_ESCRI:' = '2' AND DOC.CANC_DOC = 'N'
