[SQL]
1=SELECT MAX(REG) REG, CASE WHEN SUM(VL_REC_ATIV) > 0 THEN '0' ELSE '1' END  IND_MOV, MAX(COD_ESTAB) COD_ESTAB, MAX(RAIZ_CNPJ) RAIZ_CNPJ  FROM
2=(
3=--RECEITA DA ATIVIDADE
4=SELECT DISTINCT 'P001' REG, :COD_ESTAB COD_ESTAB, ':RAIZ_CNPJ' RAIZ_CNPJ, (DOCIT.TOTAL_DOCIT+ COALESCE(DOCIT.IPI_DOCIT,0)+COALESCE(ICMSR_DOCIT,0)+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)) VL_REC_ATIV
5=FROM DOCIT INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOCIT.CFOP_DOCIT = CFOP.COD_CFOP AND COALESCE(DOCIT.INTCFOP_DOCIT,'00') = CFOP.INT_CFOP)
6=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
7=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
8=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S","SM") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND
9=COALESCE(DOC.TPAMB_DOC,"0") <> '2' AND COALESCE(DOC.STATUS_DOC,"N") <> "I" AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
10=DOC.COD_AIDF >=0 AND DOCIT.CFOP_DOCIT < 7000 AND DOC.COD_MOD <> '57' AND 
11=SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ'
12=AND DOC.EMISS_DOC >=  :DataHora_Inicial AND DOC.EMISS_DOC <= :DataHora_Final
13=AND COALESCE(NCMIT.SPEDINSS_TRIB,'N') = 'S'
14=UNION ALL
15=SELECT DISTINCT 'P001' REG, 1 COD_ESTAB, '27259308' RAIZ_CNPJ, DOC.TOTALG_DOC VL_REC_ATIV
16=FROM DOC INNER JOIN CFOP ON (DOC.CFOPS_DOC = CFOP.COD_CFOP AND '00' = CFOP.INT_CFOP)
17=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) 
18=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = '99999999' AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
19=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ('S','SM') AND DOC.PRINT_DOC = 'S' AND COALESCE(CFOP.FIN_CFOP,'N') = 'S' AND COALESCE(CFOP.ESTAT_CFOP, 'OUTROS') = 'VENDAS' AND
20=COALESCE(DOC.TPAMB_DOC,'0') <> '2' AND COALESCE(DOC.STATUS_DOC,'N') <> 'I' AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
21=DOC.COD_AIDF >=0 AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ' AND DOC.COD_MOD = '57'
22=AND DOC.EMISS_DOC >=  :DataHora_Inicial AND DOC.EMISS_DOC <= :DataHora_Final
23=AND COALESCE(NCMIT.SPEDINSS_TRIB,'N') = 'S'
24=UNION ALL
25=SELECT DISTINCT 'P001' REG, :COD_ESTAB COD_ESTAB, ':RAIZ_CNPJ' RAIZ_CNPJ, (CUPOMIT.TOTAL + CUPOMIT.DACUPOM) VL_REC_ATIV FROM CUPOMIT
26=INNER JOIN CUPOM ON (CUPOM.COD_CF = CUPOMIT.COD_CF)
27=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOM.COD_ESTAB)
28=INNER JOIN PROD ON (CUPOMIT.COD_ESTQ = PROD.COD_ESTQ)
29=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
30=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = CUPOM.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
31=WHERE COALESCE(CUPOMIT.CANCELADO,'N') = 'N' AND CUPOMIT.DATA >=  :DataHora_Inicial AND CUPOMIT.DATA <= :DataHora_Final  AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ'
32=AND COALESCE(NCMIT.SPEDINSS_TRIB,'N') = 'S'
33=) HAVING COUNT(*) > 0
