2=(
3=--RECEITA DA ATIVIDADE
4=SELECT DOC.COD_DOC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)+ CASE WHEN ESTAB.RECBRUTACS_ESTB = 1 THEN COALESCE(IPI_DOCIT,0) ELSE 0 END) VL_REC_ATIV, 0.0 VL_EXC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)) - COALESCE(TOTALDV_DOCIT,0) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, PROD.TIPI_PROD) COD_ATIV_ECON, COALESCE(NCMIT.CODCTB_CTA,(CASE WHEN CFOP.CODCTB_CTA = '' THEN GRPIT.CODCTB_CTA ELSE CFOP.CODCTB_CTA END)) COD_CTA
5=FROM DOCIT INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOCIT.CFOP_DOCIT = CFOP.COD_CFOP AND RIGHT('00' || COALESCE(DOCIT.INTCFOP_DOCIT,'00') ,2) = CFOP.INT_CFOP)
6=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
7=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
8=INNER JOIN GRPIT ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP) = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
9=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S", "SM") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND
10=COALESCE(DOC.TPAMB_DOC,"0") <> '2' AND COALESCE(DOC.STATUS_DOC,"N") <> "I" AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
11=DOC.COD_AIDF >=0 AND DOC.COD_MOD <> '57'
12=AND ESTAB.COD_ESTAB = :COD_ESTAB:
13=AND DOC.EMISS_DOC >=  :Data_Inicial ||' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final ||' 23:59:59'
14=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') IN ('S') OR (':TUDO:' = 'S'))
15=UNION ALL
16=SELECT DISTINCT DOC.COD_DOC, (DOC.TOTALG_DOC) VL_REC_ATIV, 0.0 VL_EXC, DOC.TOTALG_DOC VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, 0) COD_ATIV_ECON, GRPIT.CODCTB_CTA COD_CTA
17=FROM DOC LEFT JOIN DOCIT ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOC.CFOPS_DOC = CFOP.COD_CFOP AND '00' = CFOP.INT_CFOP)
18=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) 
19=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = '99999999' AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
20=INNER JOIN GRPIT ON (999 = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
21=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S", "SM") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND
22=COALESCE(DOC.TPAMB_DOC,"0") <> '2' AND COALESCE(DOC.STATUS_DOC,"N") <> "I" AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
23=DOC.COD_AIDF >=0  AND DOC.COD_MOD = '57'
24=AND ESTAB.COD_ESTAB = :COD_ESTAB:
25=AND DOC.EMISS_DOC >=  :Data_Inicial ||' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final ||' 23:59:59'
26=
27=UNION ALL
28=SELECT DOC.COD_DOC, 0.0 VL_REC_ATIV, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0) + COALESCE(TOTALDV_DOCIT,0)) VL_EXC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)) - COALESCE(TOTALDV_DOCIT,0) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, PROD.TIPI_PROD) COD_ATIV_ECON, COALESCE(NCMIT.CODCTB_CTA,(CASE WHEN CFOP.CODCTB_CTA = '' THEN GRPIT.CODCTB_CTA ELSE CFOP.CODCTB_CTA END)) COD_CTA
29=FROM DOCIT INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOCIT.CFOP_DOCIT = CFOP.COD_CFOP AND RIGHT('00' || COALESCE(DOCIT.INTCFOP_DOCIT,'00') ,2) = CFOP.INT_CFOP)
30=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
31=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
32=INNER JOIN GRPIT ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP) = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
33=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S", "SM") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND

36=AND ESTAB.COD_ESTAB = :COD_ESTAB: AND DOC.COD_MOD <> '57'
37=AND DOC.EMISS_DOC >=  :Data_Inicial ||' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final ||' 23:59:59'
38=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') IN ('S') OR (':TUDO:' = 'S'))
39=UNION ALL
40=SELECT DOC.COD_DOC, 0.0 VL_REC_ATIV, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0) + COALESCE(TOTALDV_DOCIT,0)) VL_EXC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)) - COALESCE(TOTALDV_DOCIT,0) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, 0) COD_ATIV_ECON, COALESCE(NCMIT.CODCTB_CTA,(CASE WHEN CFOP.CODCTB_CTA = '' THEN GRPIT.CODCTB_CTA ELSE CFOP.CODCTB_CTA END)) COD_CTA
41=FROM DOCIT INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOCIT.CFOP_DOCIT = CFOP.COD_CFOP AND RIGHT('00' || COALESCE(DOCIT.INTCFOP_DOCIT,'00') ,2) = CFOP.INT_CFOP)
42=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) LEFT JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
43=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = '99999999' AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
44=INNER JOIN GRPIT ON (999 = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
45=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S", "SM") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND
46=COALESCE(DOC.TPAMB_DOC,"0") <> '2' AND COALESCE(DOC.STATUS_DOC,"N") <> "I" AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
47=DOC.COD_AIDF >=0 AND (COALESCE(NCMIT.SPEDINSS_TRIB,"S") = "N" OR (DOCIT.CFOP_DOCIT > 7000)) 
48=AND ESTAB.COD_ESTAB = :COD_ESTAB: AND DOC.COD_MOD = '57'
49=AND DOC.EMISS_DOC >=  :Data_Inicial ||' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final ||' 23:59:59'
50=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') IN ('S') OR (':TUDO:' = 'S'))
51=UNION ALL
52=SELECT 0 COD_DOC, (CUPOMIT.TOTAL + CUPOMIT.DACUPOM) VL_REC_ATIV, 0.0 VL_EXC, (CUPOMIT.TOTAL + CUPOMIT.DACUPOM) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, PROD.TIPI_PROD) COD_ATIV_ECON, COALESCE(NCMIT.CODCTB_CTA,(CASE WHEN CFOP.CODCTB_CTA = '' THEN GRPIT.CODCTB_CTA ELSE CFOP.CODCTB_CTA END)) COD_CTA FROM CUPOMIT
53=INNER JOIN CUPOM ON (CUPOM.COD_CF = CUPOMIT.COD_CF)
54=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOM.COD_ESTAB)
55=INNER JOIN PROD ON (CUPOMIT.COD_ESTQ = PROD.COD_ESTQ)
56=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
57=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = CUPOM.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
58=INNER JOIN GRPIT ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP) = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP)
59=WHERE COALESCE(CUPOMIT.CANCELADO,'N') = 'N' AND CUPOMIT.DATA >=  :DataHora_Inicial AND CUPOMIT.DATA <= :DataHora_Final  AND ESTAB.COD_ESTAB = :COD_ESTAB:
60=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') IN ('S') OR (':TUDO:' = 'S')) AND (':IND_MOV:' = '0')
61=)
62=GROUP BY COD_ATIV_ECON, COD_CTA
