1=SELECT  'P200' REG, CAST(:Data_Final AS DATE) PER_REF, SUM(VL_TOT_CONT_APU) VL_TOT_CONT_APU, 0.0 VL_TOT_AJ_REDUC, 0.0 VL_TOT_AJ_ACRES, SUM(VL_TOT_CONT_APU) VL_TOT_CONT_DEV,  COD_REC FROM
2=(
3=SELECT  'P200' REG, CAST(:Data_Final AS DATE) PER_REF, (SUM(VL_BC_CONT) * MAX(ALIQ_CONT)) / 100 VL_TOT_CONT_APU, 0.0 VL_TOT_AJ_REDUC, 0.0 VL_TOT_AJ_ACRES, (SUM(VL_BC_CONT) * MAX(ALIQ_CONT)) / 100 VL_TOT_CONT_DEV,  COD_REC FROM
4=(
5=--RECEITA DA ATIVIDADE
6=SELECT NCMIT.CODREC_INSS COD_REC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0)) VL_REC_ATIV, 0.0 VL_EXC, (DOCIT.TOTAL_DOCIT+COALESCE(SEGURO_DOCIT,0)+COALESCE(OUTDESP_DOCIT,0)+COALESCE(FRETE_DOCIT,0))  - COALESCE(DOCIT.TOTALDV_DOCIT,0) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, PROD.TIPI_PROD) COD_ATIV_ECON
7=FROM DOCIT INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC) INNER JOIN CFOP ON (DOCIT.CFOP_DOCIT = CFOP.COD_CFOP AND DOCIT.INTCFOP_DOCIT = CFOP.INT_CFOP)
8=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB) INNER JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
9=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
10=WHERE DOC.COD_AIDF >= 0 AND DOC.MOV_DOC in ("S") AND DOC.PRINT_DOC = "S" AND COALESCE(CFOP.FIN_CFOP,"N") = "S" AND COALESCE(CFOP.ESTAT_CFOP, "OUTROS") = "VENDAS" AND
11=COALESCE(DOC.TPAMB_DOC,"0") <> '2' AND COALESCE(DOC.STATUS_DOC,"N") <> "I" AND (COALESCE(DOC.CANC_DOC,'N') = 'N') AND
12=DOC.COD_AIDF >=0 AND DOCIT.CFOP_DOCIT < 7000
13=AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ'
14=AND DOC.EMISS_DOC >=  :Data_Inicial ||' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final ||' 23:59:59'
15=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') = 'S' OR (':TUDO:' = 'S')) AND (COALESCE(NCMIT.ALIQ_INSS,0)) > 0
16=UNION ALL
17=SELECT NCMIT.CODREC_INSS COD_REC, (CUPOMIT.TOTAL + CUPOMIT.DACUPOM) VL_REC_ATIV, 0.0 VL_EXC, (CUPOMIT.TOTAL + CUPOMIT.DACUPOM) VL_BC_CONT, COALESCE(NCMIT.ALIQ_INSS,0) ALIQ_CONT, COALESCE(NCMIT.CODATV_INSS, PROD.TIPI_PROD) COD_ATIV_ECON FROM CUPOMIT
18=INNER JOIN CUPOM ON (CUPOM.COD_CF = CUPOMIT.COD_CF)
19=INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CUPOM.COD_ESTAB)
20=INNER JOIN PROD ON (CUPOMIT.COD_ESTQ = PROD.COD_ESTQ)
21=INNER JOIN CFOP ON (CFOP.COD_CFOP = 5102 AND CFOP.INT_CFOP = '00')
22=LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = CUPOM.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
23=WHERE COALESCE(CUPOMIT.CANCELADO,'N') = 'N' AND CUPOMIT.DATA >=  :Data_Inicial AND CUPOMIT.DATA <= :Data_Final  AND SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = ':RAIZ_CNPJ'
24=AND (COALESCE(NCMIT.SPEDINSS_TRIB,'N') = 'S' OR (':TUDO:' = 'S'))  AND ((COALESCE(NCMIT.ALIQ_INSS,0)) > 0)  AND (':IND_MOV:' = '0')
25=) MV WHERE ':IND_MOV:' = '0'
26=GROUP BY COD_REC
27=)
28=GROUP BY COD_REC
