2  = REPLACE(REPLACE(CTB_MOV.HIST_CTB, CTB_MOV.HISTP_CTB, ""), '_', '-') HIST, '' COD_PART FROM CTB_MOV 
3  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CTB_MOV.COD_ESTAB) 
4  = WHERE ESTAB.CTRLCTB_ESTB  = :CTRLCTB_ESTB AND CTB_MOV.D_COD_CTA > 0 AND CTB_MOV.CODLAN_CTB = :NUM_LCTO: AND CTB_MOV.DATA_CTB >= :Data_Inicial || ' 00:00:00' AND CTB_MOV.DATA_CTB <= :Data_Final || ' 23:59:59' AND CTB_MOV.PART_CTB IN ('D', 'DC')
5  = UNION ALL
6  = SELECT 'I250' REG, CTB_MOV.CODLAN_CTB NUM_LCTO, CTB_MOV.C_CODCTB_CTA COD_CTA, '' COD_CCUS, CTB_MOV.TOTAL_CTB VL_DOC, 'C' IND_DC, '' NUM_ARQ, CASE WHEN COALESCE(CTB_MOV.COD_HP,0) = 0 THEN "1" ELSE CTB_MOV.COD_HP END COD_HIST_PAD, 
7  = CASE WHEN COALESCE(CTB_MOV.COD_HP,0) = 0 AND CTB_MOV.HIST_CTB = '' THEN 'LANCTO. NA DATA' ELSE REPLACE(REPLACE(CTB_MOV.HIST_CTB, CTB_MOV.HISTP_CTB, ""), '_', ' ') END HIST, '' COD_PART FROM CTB_MOV
8  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CTB_MOV.COD_ESTAB) 
9  = WHERE ESTAB.CTRLCTB_ESTB  = :CTRLCTB_ESTB AND CTB_MOV.C_COD_CTA > 0 AND CTB_MOV.CODLAN_CTB = :NUM_LCTO: AND CTB_MOV.DATA_CTB >= :Data_Inicial || ' 00:00:00' AND CTB_MOV.DATA_CTB <= :Data_Final || ' 23:59:59' AND CTB_MOV.PART_CTB IN ('C', 'DC')
