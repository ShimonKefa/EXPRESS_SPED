[SQL]
1  = SELECT DISTINCT 'I050' REG, CASE WHEN CAST(COALESCE(CTB_CONTA.DTPOST_CTA,CTB_CONTA.DTINS_CTA) AS DATE) > :Data_Final THEN CAST(:Data_Inicial AS DATE) ELSE CAST(COALESCE(CTB_CONTA.DTPOST_CTA,CTB_CONTA.DTINS_CTA) AS DATE) END DT_ALT,
2  = CTB_CONTA.GRUPO_CTA COD_NAT, CTB_CONTA.TIPO_CTA IND_CTA, CTB_CONTA.NIVEL_CTA NIVEL, 
3  = CTB_CONTA.CODCTB_CTA COD_CTA, 
4  = CASE WHEN CTB_CONTA.NIVEL_CTA = 1 THEN ''
5  =      WHEN CTB_CONTA.NIVEL_CTA = 2 THEN CTB_CONTA.G1_CTA 
6  =      WHEN CTB_CONTA.NIVEL_CTA = 3 THEN CTB_CONTA.G2_CTA 
7  =      WHEN CTB_CONTA.NIVEL_CTA = 4 THEN CTB_CONTA.G3_CTA  
8  =      WHEN CTB_CONTA.NIVEL_CTA = 5 THEN CTB_CONTA.G4_CTA
9  =      ELSE CTB_CONTA.G5_CTA END COD_CTA_SUP,
10 = CTB_CONTA.NOM_CTA CTA, COALESCE(CODCTB2_CTA, '-1') CODCTB2_CTA, LEFT(CTB_CONTA.CODCTB_CTA || '                    ', 20) ORDEM FROM CTB_CONTA 
11  = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = CTB_CONTA.COD_ESTAB) 
12  = WHERE ESTAB.CTRLCTB_ESTB  = :CTRLCTB_ESTB AND COALESCE(CTB_CONTA.SIT_CTA, 'A') <> 'I' 
13 = ORDER BY CTB_CONTA.CODCTB_CTA
