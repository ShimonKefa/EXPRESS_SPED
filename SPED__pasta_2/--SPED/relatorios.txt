CASE WHEN COALESCE(VDOC.ESC_DOC,'N') IN ('S','I','M') THEN 'S' ELSE 'N' END SPED_FISCAL,
CASE WHEN VDOC.COD_AIDF > 0 THEN CASE WHEN SUBSTRING(ESTAB.CNPJCPF_ESTB FROM 1 FOR 8) = SUBSTRING(VDOC.CNPJCPF_PESS FROM 1 FOR 8) THEN 'TRANSFERENCIA' ELSE 'EMISSAO PROPRIA' END ELSE 'TERCEIROS' END TIPO_EMISSAO,

VDOCIT.QTDE_DOCIT QTDE, 
CASE WHEN VDOCIT.COD_DOCIT < 900 THEN VDOCIT.UN_DOCIT ELSE '' END UN,

CASE
WHEN VDOC.FINNFE_DOC = '2' THEN '06 - DOC.FISCAL COMPLEMENTAR'
WHEN VDOC.TIPO_DOC = 'E' THEN '01 - ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR'
WHEN VDOC.CANC_DOC = 'S' THEN '02 - DOCUMENTO CANCELADO'
WHEN VDOC.TIPO_DOC = 'E' AND VDOC.CANC_DOC = 'S' THEN '03 - ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR'
WHEN VDOC.TIPO_DOC = 'E' AND VDOC.FINNFE_DOC = '2' THEN '01 - ESCRITURACAO EXTEMPORANEA DE DOC.REGULAR'
WHEN VDOC.STATUS_DOC = 'I' THEN '05 - INUTILIZADO'
ELSE    '00 - DOCUMENTO REGULAR'
END COD_SIT,
VDOCIT.CSTICMSE_DOCIT || ' ' CST_ICMS,
VDOCIT.CSTIPIE_DOCIT || ' ' CST_IPI,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.ICMSR_DOCIT,0) + COALESCE(VDOCIT.IPI_DOCIT,0) + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0)  END TOTAL,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.BCICMSE_DOCIT END BCICMS,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE CAST(COALESCE(VDOCIT.ALIQICMS_DOCIT,0) AS NUMERIC(10,2)) END ALIQ_ICMS,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.ICMSE_DOCIT END ICMS,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.BCIPIE_DOCIT END BCIPI,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.ALIQIPI_DOCIT END ALIQ_IPI,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.IPIE_DOCIT END IPI,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.IPI_DOCIT END IPI_DOC,

 CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE COALESCE(NCMIT.ALIQ_INSS,0) END ALIQ_INSS,
CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE (VDOCIT.TOTAL_DOCIT+COALESCE(VDOCIT.ICMSR_DOCIT,0)+COALESCE(VDOCIT.SEGURO_DOCIT,0)+COALESCE(VDOCIT.OUTDESP_DOCIT,0)+COALESCE(VDOCIT.FRETE_DOCIT,0)) * COALESCE(NCMIT.ALIQ_INSS,0) / 100 END INSS,

COALESCE(GRPIT_SC.BC_COFINS,'S') CALC_BC_COFINS,

CASE WHEN VDOCIT.CFOP_DOCIT = 5929 THEN 0 ELSE VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.ICMSR_DOCIT,0) + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0)  END VL_ITEM_SPED_CONTRIB,

 0

  CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
   (VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
   ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_PIS,0) / 100)
  ELSE
   (VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
   ((VDOCIT.TOTAL_DOCIT+ COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_PIS,0) / 100) 
  END

CASE WHEN (COALESCE(GRPIT_SC.BC_PIS,'S') = 'N') OR (COALESCE(VDOCIT.PEDRESP_DOCIT,'N') = 'DEV') THEN
 0
ELSE
 CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
  ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
  ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_PIS,0) / 100)) * COALESCE(GRPIT_SC.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) / 100

  ((VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
  ((VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_PIS,0) / 100)) * COALESCE(GRPIT_SC.ALIQ_PIS, ESTAB.ALIQPIS_ESTB) / 100
 END

CASE WHEN (COALESCE(GRPIT_SC.BC_COFINS,'S') = 'N') THEN
 0
ELSE
  CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
   (VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
   ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_COFINS,0) / 100)
  ELSE
   (VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
   ((VDOCIT.TOTAL_DOCIT+ COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_COFINS,0) / 100) 
  END
END VL_BC_COFINS,

 0
ELSE
 CASE WHEN ESTAB.ATIV1_ESTB = 0 THEN
  ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
  ((VDOCIT.TOTAL_DOCIT + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_COFINS,0) / 100)) * COALESCE(GRPIT_SC.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
 ELSE
  ((VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) - (COALESCE(VDOCIT.TOTALDV_DOCIT,0)) - 
  ((VDOCIT.TOTAL_DOCIT  + COALESCE(VDOCIT.FRETE_DOCIT,0) + COALESCE(VDOCIT.OUTDESP_DOCIT,0) + COALESCE(VDOCIT.SEGURO_DOCIT,0) + COALESCE(VDOCIT.ICMSR_DOCIT,0)) * COALESCE(GRPIT_SC.RDBC_COFINS,0) / 100)) * COALESCE(GRPIT_SC.ALIQ_COFINS, ESTAB.ALIQCOFINS_ESTB) / 100
 END
END VL_COFINS,
CASE 
 WHEN ESTAB.TPESCCS_ESTB = 1 THEN '1-CONSOLIDADO'  
 WHEN ESTAB.TPESCCS_ESTB = 2 THEN '2-INDIVIDUAL'
 ELSE 
 '????????????' END TIPO_ESCR_SPED_CONTRIB,

 WHEN ESTAB.INCTRIB1_ESTB = 1 THEN '1-REGIME NAO-CUMULATIVO'  
 WHEN ESTAB.INCTRIB1_ESTB = 2 THEN '2-REGIME CUMULATIVO'
 WHEN ESTAB.INCTRIB1_ESTB = 3 THEN '3-REGIME MISTO'
ELSE 
 '????????????' END INC_TRIB_SPED_CONTRIB,  
CASE 
 WHEN ESTAB.APUCRED1_ESTB = 0 THEN '0-SEM APROP.CREDITOS'  
 WHEN ESTAB.APUCRED1_ESTB = 1 THEN '1-APROPRIIACAO DIRETA'
 WHEN ESTAB.APUCRED1_ESTB = 2 THEN '3-METODO RATEIO PROPORCIONAL'
ELSE 
 '????????????' END APU_CREDITOS_SPED_CONTRIB

INNER JOIN CFOP ON (CFOP.COD_CFOP = VDOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = VDOCIT.INTCFOP_DOCIT)
LEFT JOIN ESTAB ON (ESTAB.COD_ESTAB = VDOC.COD_ESTAB)
LEFT JOIN PROD ON (PROD.COD_ESTQ = VDOCIT.COD_ESTQ)
LEFT JOIN GRP GRP_SC ON (GRP_SC.COD_GRP = PROD.CODSC_GRP)
LEFT JOIN GRPIT GRPIT_SC ON (COALESCE(PROD.CODSC_GRP, PROD.COD_GRP) = GRPIT_SC.COD_GRP AND ESTAB.COD_ESTAB = GRPIT_SC.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT_SC.CLCTB_CFOP)
LEFT JOIN NCMIT ON (NCMIT.COD_NCM = PROD.TIPI_PROD AND ESTAB.COD_ESTAB = NCMIT.COD_ESTAB AND CFOP.CLCTB_CFOP = NCMIT.CLCTB_CFOP)
LEFT JOIN PESSOAS ON (PESSOAS.COD_PESS = VDOC.COD_PESS)
LEFT JOIN MA ON (MA.COD_MA = PROD.COD_MA)
WHERE 
(VDOC.COD_AIDF >=0) AND (VDOC.PRINT_DOC = 'S')
AND COALESCE(VDOC.CANC_DOC,'N') <> 'S'  AND 
((VDOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND VDOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND VDOC.MOV_DOC = 'S') OR (VDOC.ENTSAID_DOC >= :Data_Inicial || ' 00:00:00'  AND VDOC.ENTSAID_DOC <= :Data_Final || ' 23:59:59' AND VDOC.MOV_DOC = 'E'))
AND VDOCIT.COD_DOCIT < 900 AND
